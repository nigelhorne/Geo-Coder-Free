   34: =head1 SYNOPSIS
   35: 
   36:     use Geo::Coder::Free::Local;
   37: 
   38:     my $geocoder = Geo::Coder::Free::Local->new();
   39:     my $location = $geocoder->geocode(location => 'Ramsgate, Kent, UK');
   40:     printf "lat=%.6f lon=%.6f\n", $location->lat(), $location->long();
   41: 
   42: =head1 DESCRIPTION
   43: 
   44: Provides geocoding via a user-curated CSV dataset embedded in the module's
   45: C<__DATA__> section.  Locations in the data were verified by GPS and by
   46: inspecting geotagged photographs.  The data is read once at construction time
   47: and indexed for fast lookup.
   48: 
   49: This is the highest-priority backend tried by C<Geo::Coder::Free>.
   50: 
   51: =head1 LIMITATIONS
   52: 
   53: =over 4
   54: 
   55: =item * The embedded C<__DATA__> dataset covers only a small set of hand-picked locations.
   56: There is no mechanism for non-authors to contribute data without patching the module.
   57: 
   58: =item * Canadian and Australian address parsing is not yet implemented; those queries return C<undef>.
   59: 
   60: =item * C<_search> performs an O(n) linear scan through all rows.  The hash-based
   61: index in C<new()> handles exact string matches; no index exists for partial field matches.
   62: 
   63: =item * C<our %alternatives> duplicates mappings in C<Geo::Coder::Free::__DATA__>.
   64: Both should be consolidated into a shared external config file.
   65: 
   66: =item * C<$libpostal_is_installed> is a module-level (effectively global) flag.
   67: Not thread-safe in a forking or threaded Perl deployment.
   68: 
   69: =item * The C<__DATA__> filehandle is a one-shot resource.  The first call to
   70: C<new()> exhausts it with C<my @data = E<lt>DATAE<gt>>; every subsequent
   71: C<new()> in the same process reads zero rows and builds an empty index.
   72: Construct exactly one C<Local> object per process and share it.
   73: 
   74: =item * The hash-index key is C<lc(Geo::Location::Point-E<gt>new($row)-E<gt>as_string())>,
   75: which B<includes the C<name> field>.  To get a direct index hit the caller must
   76: supply the full string with the venue/place name as the leading component;
   77: an address that omits the name falls through to the slower O(n) C<_search> path.
   78: 
   79: =back
   80: 
   81: =cut

  116: =head3 SYNOPSIS
  117: 
  118:     my $geocoder = Geo::Coder::Free::Local->new();
  119:     my $geocoder = Geo::Coder::Free::Local->new(cache => $chi_cache);

  121: =head3 DESCRIPTION
  122: 
  123: Constructor.  Reads the C<__DATA__> CSV block, builds a hash-based lookup
  124: index, and derives the geographic centre of any city/state/country cluster
  125: containing three or more data points.
  126: 
  127: B<One-shot filehandle>: the C<DATA> handle is consumed on the first call to
  128: C<new()>.  Any subsequent call to C<new()> returns an object whose dataset and
  129: index are both empty.  In code that constructs multiple C<Local> objects (e.g.
  130: in tests) construct B<exactly one> instance and reuse it across all callers.
  131: 
  132: =head3 API SPECIFICATION
  133: 
  134: =head4 input
  135: 
  136:     # Input schema (Params::Validate::Strict)
  137:     cache => { type => 'object', optional => 1, can => ['get', 'set'] }  # CHI-compatible cache object
  138: 
  139: =head4 output
  140: 
  141:     # Output schema (Return::Set)
  142:     { type => 'object', isa => 'Geo::Coder::Free::Local' }
  143: 
  144: =head3 FORMAL SPECIFICATION
  145: 
  146:     LocalState ::= ⟨⟨ data  : seq Row;
  147:                        index : Map[STRING → Row];
  148:                        cache : Map[STRING → Point] ⟩⟩
  149: 
  150:     Init : Params → LocalState
  151:     ∀ p : Params •
  152:       let rows    == parse_csv(__DATA__) ∪ geographic_centres(__DATA__) •
  153:       let idx_key == λ r • lc(Point(r).as_string()) •
  154:       LocalState.index = { idx_key(r) ↦ r | r ∈ rows }
  155: 
  156: =cut

  195: 		my $key = lc(Geo::Location::Point->new($row)->as_string());

  197: 	}

  199: 	# Build an O(1) region existence index used by geocode() to skip the O(n)

  201: 	# Major: every (state|country) pair in the dataset is represented here.

  203: 	# Conclusion: early return is safe — eliminates the O(n) grep per call.

  205: 		my $rk = uc($row->{'state'} // '') . '|' . uc($row->{'country'} // '');
  206: 		$self->{'region_index'}{$rk} = 1;
  207: 	}
  208: 
  209: 	return $self;

  212: =head2 geocode
  213: 
  214: =head3 SYNOPSIS
  215: 
  216:     my $pt = $geocoder->geocode(location => '203 E Chatsworth Rd, Reisterstown, Baltimore, MD, US');
  217:     print $pt->lat(), "\n";
  218: 
  219:     # All calling forms are accepted:
  220:     $geocoder->geocode('203 E Chatsworth Rd, Reisterstown, MD, US');
  221:     $geocoder->geocode({ location => '203 E Chatsworth Rd, Reisterstown, MD, US' });
  222: 
  223: =head3 API SPECIFICATION
  224: 
  225: =head4 input
  226: 
  227:     # Input schema (Params::Validate::Strict)
  228:     location => { type => 'scalar' }  # address string; must contain at least two commas
  229: 
  230: =head4 output
  231: 
  232:     # Output schema (Return::Set)
  233:     { type => 'object', isa => 'Geo::Location::Point', optional => 1 }
  234: 
  235: =head3 PROVENANCE
  236: 
  237: Every result carries C<geocoder>, C<database>, C<confidence> and C<accuracy>
  238: (section 2.1 of F<BI-Phase6-Dependency-Specs.md>).  This backend reports
  239: C<database> as C<Geo::Coder::Free::Local>, and C<accuracy> is what the matched
  240: columns justify: a house number makes it an C<address>, a road a C<street>, a
  241: name or city a C<locality>, a state a C<region> and a country a C<country>.  A
  242: lookup answered from the index, which matches a whole row, is an exact match
  243: (C<confidence> of 1.0) whose accuracy comes from the row's own fields - so a
  244: derived geographic centre, which carries a city, state and country and nothing
  245: narrower, is a C<locality>.
  246: 
  247: =head3 MESSAGES
  248: 
  249:     Usage: ...::geocode(...)   No location argument given.
  250: 
  251: =head3 FORMAL SPECIFICATION
  252: 
  253:     Geocode : STRING → Point?
  254:     ∀ addr : STRING •
  255:       let norm == lc(replace_usa(addr)) •
  256:       (norm ∈ cache ⟹ result = cache[norm]) ∧
  257:       (norm ∈ index ⟹ result = index[norm]) ∧
  258:       (¬ result ⟹ result = parse_and_search(addr))
  259: 
  260: =head3 PSEUDOCODE
  261: 
  262:     normalise @_ into %params
  263:     reject if location does not contain two or more commas (not a full address)
  264:     check cache; return hit
  265:     check hash index; return hit and cache it
  266:     # index key includes the 'name' field; supply the venue/place name as the
  267:     # leading address component to guarantee an index hit
  268:     attempt country-specific parser (US, GB; skip CA/AU pending implementation)
  269:     attempt Geo::StreetAddress::US for "..., USA" addresses
  270:     attempt Geo::Address::Parser
  271:     attempt Geo::libpostal (large memory footprint; loaded lazily)
  272:     attempt 4-part regex decomposition (name/road/city/state/country)
  273:     attempt %alternatives mapping
  274:     return undef
  275: 
  276: =cut

  302: 
  303: 	# This backend only handles full addresses (at least "road, city, country")
  304: 	return if $location !~ /,.+,/;
  305: 
  306: 	# Normalise the lookup key — "USA" is treated the same as "US"
  307: 	my $lc = lc($location);
  308: 	$lc =~ s/,\s*usa$/, us/i;

  310: 	return $self->{'cache'}{$lc}  if exists $self->{'cache'}{$lc};

  312: 		if exists $self->{'index'}{$lc};

  314: 	my $ap;

  316: 		$ap = $self->{'ap'}{'us'} //= Lingua::EN::AddressParse->new(
  317: 			country => 'US', auto_clean => 1, force_case => 1, force_post_code => 0

  320: 		$ap = $self->{'ap'}{'gb'} //= Lingua::EN::AddressParse->new(
  321: 			country => 'GB', auto_clean => 1, force_case => 1, force_post_code => 0
  322: 		);
  323: 	} elsif ($location =~ /Canada$/) {
  324: 		# TODO: Canadian address parsing not yet implemented
  325: 		return;
  326: 	} elsif ($location =~ /Australia$/) {
  327: 		# TODO: Australian address parsing not yet implemented
  328: 		return;
  329: 	}

  348: 			}

  493: 				return $self->_cache_and_return($lc, $rc);
  494: 			}
  495: 		}
  496: 	}
  497: 
  498: 	# Geo::libpostal — accurate but uses enormous RAM; initialise at most once
  499: 	if ($libpostal_is_installed == LIBPOSTAL_UNKNOWN) {
  500: 		if (eval { require Geo::libpostal; 1 }) {
  501: 			Geo::libpostal->import();
  502: 			$libpostal_is_installed = LIBPOSTAL_INSTALLED;
  503: 		} else {
  504: 			$libpostal_is_installed = LIBPOSTAL_NOT_INSTALLED;
  505: 		}
  506: 	}
  507: 
  508: 	if ($libpostal_is_installed == LIBPOSTAL_INSTALLED) {
  509: 		my %addr = Geo::libpostal::parse_address($location);
  510: 		if (%addr) {
  511: 			# Normalise field names to match our data schema
  512: 			$addr{'number'} = delete $addr{'house_number'} if $addr{'house_number'} && !$addr{'number'};
  513: 			$addr{'name'}   = delete $addr{'house'}        if $addr{'house'}        && !$addr{'name'};
  514: 			$addr{'location'} = $location;
  515: 
  516: 			if (my $street = $addr{'road'}) {
  517: 				$addr{'road'} = _normalize($street);
  518: 			}
  519: 
  520: 			# libpostal returns "england" as a state — map to country code
  521: 			if (defined($addr{'state'}) && !defined($addr{'country'})
  522: 			    && $addr{'state'} eq 'england') {
  523: 				delete $addr{'state'};
  524: 				$addr{'country'} = 'GB';
  525: 			}
  526: 
  527: 			if ($addr{'country'} && ($addr{'state'} || $addr{'state_district'})) {
  528: 				if ($addr{'country'} =~ /Canada/i) {
  529: 					$addr{'country'} = 'Canada';
  530: 					$addr{'state'}   = _to_two_letter_state('Canada', $addr{'state'})
  531: 						if $addr{'state'} && length($addr{'state'}) > 2;
  532: 				} elsif ($addr{'country'} =~ /^(United States|USA|US)$/i) {
  533: 					$addr{'country'} = 'US';
  534: 					$addr{'state'}   = _to_two_letter_state('US', $addr{'state'})
  535: 						if $addr{'state'} && length($addr{'state'}) > 2;
  536: 				}
  537: 
  538: 				if ($addr{'state_district'}) {
  539: 					$addr{'state_district'} =~ s/^(.+)\s+COUNTY\s*$/$1/i;
  540: 					if (my $rc = $self->_search(\%addr, qw(number road city state_district state country))) {
  541: 						return $self->_cache_and_return($lc, $rc);
  542: 					}
  543: 				}
  544: 				if (my $rc = $self->_search(\%addr, qw(number road city state country))) {
  545: 					return $self->_cache_and_return($lc, $rc);
  546: 				}
  547: 				if ($addr{'number'}) {
  548: 					if (my $rc = $self->_search(\%addr, qw(road city state country))) {
  549: 						return $self->_cache_and_return($lc, $rc);
  550: 					}
  551: 				}
  552: 			}
  553: 		}
  554: 	}
  555: 
  556: 	# Last resort: decompose "road, city, state, country" with a regex
  557: 	if ($location =~ /^(.+?),\s*([\s\w]+),\s*([\s\w]+),\s*([\w\s]+)$/) {
  558: 		my %addr = (
  559: 			road    => $1,
  560: 			city    => $2,
  561: 			state   => $3,
  562: 			country => $4,
  563: 		);
  564: 		$addr{'state'}   =~ s/\s+$//g;
  565: 		$addr{'country'} =~ s/\s+$//g;
  566: 
  567: 		if ($addr{'road'} =~ /([\w\s]+),*\s+(.+)/) {
  568: 			$addr{'name'} = $1;
  569: 			$addr{'road'} = $2;
  570: 		}
  571: 		if ($addr{'road'} =~ /^(\d+)\s+(.+)/) {
  572: 			$addr{'number'} = $1;
  573: 			$addr{'road'}   = $2;
  574: 			if (my $rc = $self->_search(\%addr, qw(name number road city state country))) {
  575: 				return $self->_cache_and_return($lc, $rc);
  576: 			}
  577: 		} elsif (my $rc = $self->_search(\%addr, qw(name road city state country))) {
  578: 			return $self->_cache_and_return($lc, $rc);
  579: 		}
  580: 		if ($addr{'name'} && !defined($addr{'number'})) {
  581: 			if (my $rc = $self->_search(\%addr, qw(name road city state country))) {
  582: 				return $self->_cache_and_return($lc, $rc);
  583: 			}
  584: 		}
  585: 	}
  586: 
  587: 	# Try the alternatives table — curated mappings for inconsistent names
  588: 	$location = uc($location);
  589: 	for my $left (keys %alternatives) {
  590: 		next unless $location =~ $left;
  591: 		(my $mapped = $location) =~ s/$left/$alternatives{$left}/;
  592: 		$params{'location'} = $mapped;
  593: 		if (my $rc = $self->geocode(\%params)) {
  594: 			return $self->_cache_and_return($lc, $rc);
  595: 		}
  596: 		if ($mapped =~ /(.+), (England|UK)$/i) {
  597: 			$params{'location'} = "$1, GB";
  598: 			if (my $rc = $self->geocode(\%params)) {
  599: 				return $self->_cache_and_return($lc, $rc);
  600: 			}
  601: 		}
  602: 		$params{'location'} = $location;
  603: 	}
  604: 
  605: 	return;
  606: }

  608: =head2 reverse_geocode
  609: 
  610: =head3 SYNOPSIS
  611: 
  612:     my $loc = $geocoder->reverse_geocode(latlng => '51.3341,-1.4159');
  613:     # Returns the location string(s) for that lat/lon pair.
  614: 
  615: =head3 API SPECIFICATION
  616: 
  617: =head4 input
  618: 
  619:     # Input schema (Params::Validate::Strict) — latlng or lat+lon required
  620:     latlng => { type => 'scalar', optional => 1 }  # "$lat,$long" comma-separated decimal degrees
  621:     lat    => { type => 'scalar', optional => 1 }  # latitude  (alternative to latlng)
  622:     lon    => { type => 'scalar', optional => 1 }  # longitude (alternative to latlng)
  623:     long   => { type => 'scalar', optional => 1 }  # alias for lon
  624: 
  625: =head4 output
  626: 
  627:     # Output schema (Return::Set)
  628:     # scalar context: { type => 'scalar',   optional => 1 }  # location string
  629:     # list context:   { type => 'arrayref', of => { type => 'scalar' } }
  630: 
  631: =cut

  633: sub reverse_geocode {
  634: 	my $self = shift;
  635: 
  636: 	if (!ref($self)) {
  637: 		if (scalar @_) {
  638: 			return __PACKAGE__->new()->reverse_geocode(@_);
  639: 		} elsif (!defined($self)) {
  640: 			Carp::croak('Usage: ', __PACKAGE__, '::reverse_geocode(latlng => "$lat,$long")');
  641: 		} elsif ($self eq __PACKAGE__) {
  642: 			Carp::croak("Usage: $self", '::reverse_geocode(latlng => "$lat,$long")');
  643: 		}
  644: 		return __PACKAGE__->new()->reverse_geocode($self);
  645: 	} elsif (ref($self) eq 'HASH') {
  646: 		return __PACKAGE__->new()->reverse_geocode($self);
  647: 	}
  648: 
  649: 	my %params = _normalize_args(
  650: 		'Usage: ' . __PACKAGE__ . '::reverse_geocode(latlng => "$lat,$long")',
  651: 		'latlng', @_
  652: 	);
  653: 
  654: 	my ($latitude, $longitude);
  655: 	if (my $latlng = $params{'latlng'}) {
  656: 		($latitude, $longitude) = split /,/, $latlng;
  657: 	} else {
  658: 		$latitude  = $params{'lat'};
  659: 		$longitude = $params{'lon'} // $params{'long'};
  660: 	}
  661: 
  662: 	if (!defined($latitude) || !defined($longitude)) {
  663: 		Carp::croak('Usage: ', __PACKAGE__, '::reverse_geocode(latlng => "$lat,$long")');
  664: 	}
  665: 
  666: 	my @rc;
  667: 	for my $row (@{$self->{'data'}}) {
  668: 		next unless defined($row->{'latitude'}) && defined($row->{'longitude'});
  669: 		next unless _equal($row->{'latitude'}, $latitude, 4)
  670: 		         && _equal($row->{'longitude'}, $longitude, 4);
  671: 
  672: 		# Rows in $self->{'data'} are blessed as Geo::Location::Point by
  673: 		# the index-building loop in new() — calling as_string() is safe here.
  674: 		my $location = uc($row->as_string());
  675: 		if (wantarray) {
  676: 			push @rc, $location;
  677: 			# Add any reverse-alternative mappings for this location
  678: 			while (my ($left, $right) = each %alternatives) {
  679: 				if ($location =~ $right) {
  680: 					(my $l = $location) =~ s/$right/$left/;
  681: 					push @rc, $l;
  682: 				}
  683: 			}
  684: 		} else {
  685: 			return $location;
  686: 		}
  687: 	}
  688: 	return @rc;
  689: }

  697: ◆sub ua { }  698: ◆  699: ◆# -----------------------------------------------------------------------  700: ◆# Private helpers  701: ◆# -----------------------------------------------------------------------  702: ◆  703: ◆# Purpose:  Normalise the four calling conventions used by geocode/reverse_geocode:  704: ◆#           hashref, even-length key-val list, single bare string.  705: ◆# Entry:    $error_msg  — croak message if an unsupported ref type is passed.  706: ◆#           $bare_key   — hash key to use when a single bare string is given  707: #                         ('location' for geocode, 'latlng' for reverse_geocode).
  708: #           @args       — raw @_ after $self has been shifted.
  709: # Exit:     Flat %params hash.
  710: # Side Effects: None; croaks on non-hash reference arguments.
  711: sub _normalize_args {
  712: 	my ($error_msg, $bare_key, @args) = @_;
  713: 	return %{$args[0]}               if ref($args[0]) eq 'HASH';
  714: 	Carp::croak($error_msg)          if ref($args[0]);
  715: 	return @args                     if @args && @args % 2 == 0;
  716: 	return ($bare_key => $args[0])   if @args == 1;
  717: 	# M7 (logic-gap closure): odd-count > 1 is not a valid calling convention.
  718: 	# Fail fast rather than silently discarding trailing arguments.
  719: 	Carp::croak($error_msg)          if @args;
  720: 	return ();
  721: }

  723: # Purpose:  Store $rc in the lookup cache under $key and return it.
  724: #           Reduces the repeated "$self->{cache}{$k} = $r; return $r;" pattern
  725: #           that appeared ~12 times across geocode.
  726: # Entry:    $self, $key (lc string), $rc (Geo::Location::Point or undef).
  727: # Exit:     $rc (returned transparently).
  728: # Side Effects: Writes to $self->{cache}.
  729: sub _cache_and_return {
  730: 	my ($self, $key, $rc) = @_;
  731: 	$self->{'cache'}{$key} = $rc;
  732: 	return $rc;
  733: }

  735: # Purpose:  Resolve a full state/province name to its two-letter code using
  736: #           the appropriate Locale:: module.  Returns the original string if
  737: #           no mapping is found or the input is already two characters.
  738: #           Locale::US and Locale::CA objects are cached as module-level
  739: #           singletons to avoid repeated construction overhead.
  740: # Entry:    $country — country name or ISO code; $state — state/province name.
  741: # Exit:     Two-letter code or $state unchanged.
  742: # Side Effects: May initialise $_locale_us or $_locale_ca singletons.
  743: sub _to_two_letter_state {
  744: 	my ($country, $state) = @_;
  745: 	return $state // '' if !defined($state) || length($state) <= 2;
  746: 
  747: 	if ($country =~ /^(United States|USA|US)$/i) {
  748: 		$_locale_us //= Locale::US->new();
  749: 		return $_locale_us->{'state2code'}{uc($state)} // $state;
  750: 	} elsif ($country =~ /Canada/i) {
  751: 		$_locale_ca //= Locale::CA->new();
  752: 		return $_locale_ca->{'province2code'}{uc($state)} // $state;
  753: 	}
  754: 	return $state;
  755: }

  757: # Purpose:  Match parsed address components against all data rows.
  758: #           Each named column must match (case-insensitively) the corresponding
  759: #           data field; a minimum of 3 columns must match for a result.
  760: # Entry:    $self, $data — hashref of address components,
  761: #           @columns — ordered list of keys to compare.
  762: # Exit:     Geo::Location::Point on match; undef otherwise.
  763: # Side Effects: May delete undef entries from $data (for undefined optional fields).
  764: sub _search {
  765: 	my ($self, $data, @columns) = @_;
  766: 
  767: 	# M5 (boolean reduction): the $failed flag was set-and-break in the inner loop
  768: 	# then tested in the outer loop — eliminated by using a labeled 'next ROW'
  769: 	# that short-circuits directly, removing one boolean gate per iteration.
  770: 	ROW: for my $row (@{$self->{'data'}}) {
  771: 		my $matched = 0;
  772: 		my @matched_columns;
  773: 
  774: 		for my $column (@columns) {
  775: 			if (defined($data->{$column})) {
  776: 				next ROW unless defined($row->{$column});
  777: 				next ROW if uc($row->{$column}) ne uc($data->{$column});
  778: 				$matched++;
  779: 				# The accuracy is reported from the columns that matched, not
  780: 				# every column this call site offered: a caller trying
  781: 				# 'number road city...' for a town that has no road must not have
  782: 				# its match described as an address.
  783: 				push @matched_columns, $column;
  784: 			} elsif (exists $data->{$column}) {
  785: 				delete $data->{$column};
  786: 			}
  787: 		}
  788: 
  789: 		next if $matched < 3;
  790: 
  791: 		# Assign confidence based on how many columns contributed to the match
  792: 		my $confidence = $matched == scalar(@columns) ? $CONF_EXACT
  793: 		               : $matched >= 4                ? $CONF_HIGH
  794: 		               :                               $CONF_MEDIUM;
  795: 
  796: 		# Provenance (Phase 6 dependency specs section 2.1).  Set after %{$row}
  797: 		# so that a dataset column that happens to be called accuracy cannot stand
  798: 		# in for the match this code actually made.
  799: 		return Geo::Location::Point->new(
  800: 			location   => $data->{'location'},
  801: 			confidence => $confidence,
  802: 			database   => __PACKAGE__,
  803: 			%{$row},
  804: 			geocoder   => GEOCODER,
  805: 			accuracy   => _accuracy_for(@matched_columns),
  806: 		);
  807: 	}
  808: 	return;
  809: }

  811: =head2 _result_for
  812: 
  813: Build the result for a dataset row that the index matched.  The index holds the
  814: rows themselves - the same hashrefs C<_search> scans - so the object is built
  815: as a copy rather than blessed in place, and the provenance keys every result
  816: carries (section 2.1 of F<BI-Phase6-Dependency-Specs.md>) are added where the
  817: row does not already record them.
  818: 
  819: An index hit is an exact match on the row's string, so the confidence is
  820: C<$CONF_EXACT>, and the accuracy is what the row's own fields justify, through
  821: C<_accuracy_for()>.
  822: 
  823: =cut
  824: 
  825: sub _result_for {
  826: 	my ($self, $row) = @_;
  827: 
  828: 	my %attributes = %{$row};
  829: 	$attributes{'geocoder'}   = GEOCODER;
  830: 	$attributes{'database'}   = __PACKAGE__;
  831: 	$attributes{'confidence'} = $CONF_EXACT;
  832: 	$attributes{'accuracy'}   = _accuracy_for(keys %attributes);
  833: 
  834: 	return Geo::Location::Point->new(\%attributes);
  835: }

  837: # Purpose:  Calculate the geographic centres (centroids) of all city/state/country
  838: #           clusters in the parsed dataset that contain three or more data points.
  839: #           Clusters with fewer than 3 entries are too sparse to be meaningful.
  840: # Entry:    $rows — arrayref of hashrefs with latitude/longitude/city/state/country.
  841: # Exit:     Arrayref of hashrefs (one per qualifying cluster), or undef if none.
  842: # Side Effects: None.
  843: sub _find_geographic_centres {
  844: 	my $rows = $_[0];
  845: 
  846: 	# Group rows by normalised city|state|country key
  847: 	my %groups;
  848: 	for my $row (@{$rows}) {
  849: 		next unless defined($row->{'latitude'})  && defined($row->{'longitude'});
  850: 		next unless $row->{'latitude'}  =~ /^-?\d+\.?\d*$/;
  851: 		next unless $row->{'longitude'} =~ /^-?\d+\.?\d*$/;
  852: 		my $key = join '|',
  853: 			$row->{'city'}    // '',
  854: 			$row->{'state'}   // '',
  855: 			$row->{'country'} // '';
  856: 		push @{$groups{$key}}, $row;
  857: 	}
  858: 
  859: 	my @centres;
  860: 	for my $key (keys %groups) {
  861: 		my $locs = $groups{$key};
  862: 		next if @{$locs} < 3;
  863: 
  864: 		my ($city, $state, $country) = split /\|/, $key;
  865: 		my ($lat, $lon) = _calculate_centre($locs);
  866: 
  867: 		push @centres, {
  868: 			city      => $city,
  869: 			state     => $state,
  870: 			country   => $country,
  871: 			lat       => $lat,
  872: 			latitude  => $lat,
  873: 			longitude => $lon,
  874: 			long      => $lon,
  875: 			lng       => $lon,
  876: 		};
  877: 	}
  878: 
  879: 	return @centres ? \@centres : undef;
  880: }

  882: # Purpose:  Compute the arithmetic mean of latitude and longitude for a group
  883: #           of locations.  Accurate for small geographic areas; for large areas
  884: #           or locations crossing the antimeridian, a vector-mean is needed.
  885: # Entry:    $locs — arrayref of hashrefs each with 'latitude' and 'longitude'.
  886: # Exit:     ($centre_lat, $centre_lon) as decimal degree strings to 6dp.
  887: # Side Effects: None.
  888: sub _calculate_centre {
  889: 	my $locs  = $_[0];
  890: 	my ($sum_lat, $sum_lon) = (0, 0);
  891: 	$sum_lat += $_->{'latitude'}, $sum_lon += $_->{'longitude'} for @{$locs};
  892: 	my $n = scalar @{$locs};
  893: 	return (sprintf('%.6f', $sum_lat / $n), sprintf('%.6f', $sum_lon / $n));
  894: }
  895: 
  896: # https://www.oreilly.com/library/view/perl-cookbook/1565922433/ch02s03.html
  897: # Equal within $dp decimal places — avoids floating-point equality pitfalls.
  898: sub _equal {
  899: 	my ($A, $B, $dp) = @_;
  900: 	return sprintf("%.${dp}g", $A) eq sprintf("%.${dp}g", $B);
  901: }
