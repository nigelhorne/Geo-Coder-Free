   35: =head1 SYNOPSIS
   36: 
   37:     use Geo::Coder::Free::Local;
   38: 
   39:     my $geocoder = Geo::Coder::Free::Local->new();
   40:     my $location = $geocoder->geocode(location => 'Ramsgate, Kent, UK');
   41:     printf "lat=%.6f lon=%.6f\n", $location->lat(), $location->long();
   42: 
   43: =head1 DESCRIPTION
   44: 
   45: Provides geocoding via a user-curated CSV dataset embedded in the module's
   46: C<__DATA__> section.  Locations in the data were verified by GPS and by
   47: inspecting geotagged photographs.  The data is read once at construction time
   48: and indexed for fast lookup.
   49: 
   50: This is the highest-priority backend tried by C<Geo::Coder::Free>.
   51: 
   52: =head1 LIMITATIONS
   53: 
   54: =over 4
   55: 
   56: =item * The embedded C<__DATA__> dataset covers only a small set of hand-picked locations.
   57: There is no mechanism for non-authors to contribute data without patching the module.
   58: 
   59: =item * Canadian and Australian address parsing is not yet implemented; those queries return C<undef>.
   60: 
   61: =item * C<_search> performs an O(n) linear scan through all rows.  The hash-based
   62: index in C<new()> handles exact string matches; no index exists for partial field matches.
   63: 
   64: =item * C<our %alternatives> duplicates mappings in C<Geo::Coder::Free::__DATA__>.
   65: Both should be consolidated into a shared external config file.
   66: 
   67: =item * C<$libpostal_is_installed> is a module-level (effectively global) flag.
   68: Not thread-safe in a forking or threaded Perl deployment.
   69: 
   70: =item * The C<__DATA__> filehandle is a one-shot resource.  The first call to
   71: C<new()> exhausts it with C<my @data = E<lt>DATAE<gt>>; every subsequent
   72: C<new()> in the same process reads zero rows and builds an empty index.
   73: Construct exactly one C<Local> object per process and share it.
   74: 
   75: =item * The hash-index key is C<lc(Geo::Location::Point-E<gt>new($row)-E<gt>as_string())>,
   76: which B<includes the C<name> field>.  To get a direct index hit the caller must
   77: supply the full string with the venue/place name as the leading component;
   78: an address that omits the name falls through to the slower O(n) C<_search> path.
   79: 
   80: =back
   81: 
   82: =cut

  122: =head3 DESCRIPTION
  123: 
  124: Constructor.  Reads the C<__DATA__> CSV block, builds a hash-based lookup
  125: index, and derives the geographic centre of any city/state/country cluster
  126: containing three or more data points.
  127: 
  128: B<One-shot filehandle>: the C<DATA> handle is consumed on the first call to
  129: C<new()>.  Any subsequent call to C<new()> returns an object whose dataset and
  130: index are both empty.  In code that constructs multiple C<Local> objects (e.g.
  131: in tests) construct B<exactly one> instance and reuse it across all callers.
  132: 
  133: =head3 API SPECIFICATION
  134: 
  135: =head4 input
  136: 
  137:     # Input schema (Params::Validate::Strict)
  138:     cache => { type => 'object', optional => 1, can => ['get', 'set'] }  # CHI-compatible cache object
  139: 
  140: =head4 output
  141: 
  142:     # Output schema (Return::Set)
  143:     { type => 'object', isa => 'Geo::Coder::Free::Local' }
  144: 
  145: =head3 FORMAL SPECIFICATION
  146: 
  147:     LocalState ::= ⟨⟨ data  : seq Row;
  148:                        index : Map[STRING → Row];
  149:                        cache : Map[STRING → Point] ⟩⟩
  150: 
  151:     Init : Params → LocalState
  152:     ∀ p : Params •
  153:       let rows    == parse_csv(__DATA__) ∪ geographic_centres(__DATA__) •
  154:       let idx_key == λ r • lc(Point(r).as_string()) •
  155:       LocalState.index = { idx_key(r) ↦ r | r ∈ rows }
  156: 
  157: =cut

  161: 	my $params = Params::Get::get_params(undef, @_) // {};
  162: 
  163: 	if (!defined($class)) {
  164: 		$class = __PACKAGE__;	# FIXME: only works when no arguments are given
  165: 	} elsif (ref($class)) {
  166: 		return bless { %{$class}, %{$params} }, ref($class);

  169: 	$params = Object::Configure::configure($class, $params);
  170: 
  171: 	my @data = <DATA>;
  172: 
  173: 	my $self = bless {
  174: 		data => xsv_slurp(
  175: 			shape    => 'aoh',
  176: 			text_csv => {
  177: 				allow_loose_quotes => 1,
  178: 				blank_is_undef     => 1,
  179: 				empty_is_undef     => 1,

  186: 	}, $class;

  188: 	# Derive geographic centres from clusters of 3+ co-located data points,
  189: 	# adding implicit town/city records to the dataset.
  190: 	my $towns = _find_geographic_centres($self->{'data'});

  193: 	# Build a hash index on the stringified Geo::Location::Point representation
  194: 	# so that exact-match lookups avoid the O(n) scan in _search.
  195: 	for my $row (@{$self->{'data'}}) {
  196: 		my $key = lc(Geo::Location::Point->new($row)->as_string());
  197: 		$self->{'index'}{$key} = $row;
  198: 	}
  199: 
  200: 	# Build an O(1) region existence index used by geocode() to skip the O(n)
  201: 	# grep before _search when no rows for the requested state/country exist.
  202: 	# Major: every (state|country) pair in the dataset is represented here.
  203: 	# Minor: if geocode's target pair is absent, _search can never match.
  204: 	# Conclusion: early return is safe — eliminates the O(n) grep per call.
  205: 	for my $row (@{$self->{'data'}}) {
  206: 		my $rk = uc($row->{'state'} // '') . '|' . uc($row->{'country'} // '');
  207: 		$self->{'region_index'}{$rk} = 1;
  208: 	}
  209: 
  210: 	return $self;
  211: }

  251: =head2 geocode
  252: 
  253: =head3 SYNOPSIS
  254: 
  255:     my $pt = $geocoder->geocode(location => '203 E Chatsworth Rd, Reisterstown, Baltimore, MD, US');
  256:     print $pt->lat(), "\n";
  257: 
  258:     # All calling forms are accepted:
  259:     $geocoder->geocode('203 E Chatsworth Rd, Reisterstown, MD, US');
  260:     $geocoder->geocode({ location => '203 E Chatsworth Rd, Reisterstown, MD, US' });
  261: 
  262: =head3 API SPECIFICATION
  263: 
  264: =head4 input
  265: 
  266:     # Input schema (Params::Validate::Strict)
  267:     location => { type => 'scalar' }  # address string; must contain at least two commas
  268: 
  269: =head4 output
  270: 
  271:     # Output schema (Return::Set)
  272:     { type => 'object', isa => 'Geo::Location::Point', optional => 1 }
  273: 
  274: =head3 PROVENANCE
  275: 
  276: Every result carries C<geocoder>, C<database>, C<confidence> and C<accuracy>
  277: (section 2.1 of F<BI-Phase6-Dependency-Specs.md>).  This backend reports
  278: C<database> as C<Geo::Coder::Free::Local>, and C<accuracy> is what the matched
  279: columns justify: a house number makes it an C<address>, a road a C<street>, a
  280: name or city a C<locality>, a state a C<region> and a country a C<country>.  A
  281: lookup answered from the index, which matches a whole row, is an exact match
  282: (C<confidence> of 1.0) whose accuracy comes from the row's own fields - so a
  283: derived geographic centre, which carries a city, state and country and nothing
  284: narrower, is a C<locality>.
  285: 
  286: =head3 MESSAGES
  287: 
  288:     Usage: ...::geocode(...)   No location argument given.
  289: 
  290: =head3 FORMAL SPECIFICATION
  291: 
  292:     Geocode : STRING → Point?
  293:     ∀ addr : STRING •
  294:       let norm == lc(replace_usa(addr)) •
  295:       (norm ∈ cache ⟹ result = cache[norm]) ∧
  296:       (norm ∈ index ⟹ result = index[norm]) ∧
  297:       (¬ result ⟹ result = parse_and_search(addr))
  298: 
  299: =head3 PSEUDOCODE
  300: 
  301:     normalise @_ into %params
  302:     reject if location does not contain two or more commas (not a full address)
  303:     check cache; return hit
  304:     check hash index; return hit and cache it
  305:     # index key includes the 'name' field; supply the venue/place name as the
  306:     # leading address component to guarantee an index hit
  307:     attempt country-specific parser (US, GB; skip CA/AU pending implementation)
  308:     attempt Geo::StreetAddress::US for "..., USA" addresses
  309:     attempt Geo::Address::Parser
  310:     attempt Geo::libpostal (large memory footprint; loaded lazily)
  311:     attempt 4-part regex decomposition (name/road/city/state/country)
  312:     attempt %alternatives mapping
  313:     return undef
  314: 
  315: =cut

  317: sub geocode {
  318: 	my $self = shift;
  319: 
  320: 	# Handle being called as a function rather than a method
  321: 	if (!ref($self)) {
  322: 		if (scalar @_) {
  323: 			return __PACKAGE__->new()->geocode(@_);
  324: 		} elsif (!defined($self)) {
  325: 			Carp::croak('Usage: ', __PACKAGE__, '::geocode(location => $location)');
  326: 		} elsif ($self eq __PACKAGE__) {
  327: 			Carp::croak("Usage: $self", '::geocode(location => $location)');
  328: 		}
  329: 		return __PACKAGE__->new()->geocode($self);
  330: 	} elsif (ref($self) eq 'HASH') {
  331: 		return __PACKAGE__->new()->geocode($self);
  332: 	}
  333: 
  334: 	my %params = _normalize_args(
  335: 		'Usage: ' . __PACKAGE__ . '::geocode(location => $location)',
  336: 		'location', @_
  337: 	);
  338: 
  339: 	my $location = $params{'location'}
  340: 		or Carp::croak('Usage: geocode(location => $location)');
  341: 
  342: 	# Normalise the lookup key — "USA" is treated the same as "US"
  343: 	my $lc = lc($location);
  344: 	$lc =~ s/,\s*usa$/, us/i;
  345: 
  346: 	return $self->{'cache'}{$lc}  if exists $self->{'cache'}{$lc};
  347: 	return $self->_cache_and_return($lc, $self->_result_for($self->{'index'}{$lc}))
  348: 		if exists $self->{'index'}{$lc};
  349: 
  350: 	# A miss found by an earlier instance, or an earlier call through this
  351: 	# one, means the same search would find nothing again: the data does not
  352: 	# change while the process runs.  clear_miss_cache() is for when it has.
  353: 	return if $self->_missed($lc);
  354: 
  355: 	# This backend only handles full addresses (at least "road, city, country")
  356: 	return $self->_remember_miss($lc) if $location !~ /,.+,/;
  357: 
  358: 	my $ap;
  359: 	if ($location =~ /USA?$/ || $location =~ /United States$/) {
  360: 		$ap = $self->{'ap'}{'us'} //= Lingua::EN::AddressParse->new(
  361: 			country => 'US', auto_clean => 1, force_case => 1, force_post_code => 0
  362: 		);
  363: 	} elsif ($location =~ /(England|Scotland|Wales|Northern Ireland|UK|GB)$/i) {
  364: 		$ap = $self->{'ap'}{'gb'} //= Lingua::EN::AddressParse->new(
  365: 			country => 'GB', auto_clean => 1, force_case => 1, force_post_code => 0
  366: 		);
  367: 	} elsif ($location =~ /Canada$/) {
  368: 		# TODO: Canadian address parsing not yet implemented
  369: 		return $self->_remember_miss($lc);
  370: 	} elsif ($location =~ /Australia$/) {
  371: 		# TODO: Australian address parsing not yet implemented
  372: 		return $self->_remember_miss($lc);
  373: 	}
  374: 
  375: 	if ($ap) {
  376: 		my $l = $location;
  377: 		$l =~ s/(.+), (England|UK)$/$1, GB/i;
  378: 
  379: 		if ($ap->parse($l) == 0) {
  380: 			my %c = $ap->components();
  381: 			my %addr = (location => $l);
  382: 
  383: 			if (my $street = $c{'street_name'}) {
  384: 				if (my $type = $c{'street_type'}) {
  385: 					my $abbrev = _abbreviate($type);
  386: 					$street .= ' ' . ($abbrev || $type);
  387: 					$street .= ' ' . $c{'street_direction_suffix'}
  388: 						if $c{'street_direction_suffix'};
  389: 					$street =~ s/^0+//;
  390: 					$addr{'road'} = $street;
  391: 				}
  392: 			}
  393: 
  394: 			if (length($c{'subcountry'}) == 2) {
  395: 				$addr{'state'} = $c{'subcountry'};
  396: 			} else {
  397: 				my $country_raw = $c{'country'} // '';
  398: 				if ($country_raw =~ /Canada/i) {
  399: 					$addr{'country'} = 'CA';
  400: 					$addr{'state'}   = _to_two_letter_state('Canada', $c{'subcountry'});
  401: 				} elsif ($country_raw =~ /^(United States|USA|US)$/i) {
  402: 					$addr{'country'} = 'US';
  403: 					$addr{'state'}   = _to_two_letter_state('US', $c{'subcountry'});
  404: 				} elsif ($country_raw) {
  405: 					$addr{'country'} = $country_raw;
  406: 					$addr{'state'}   = $c{'subcountry'} if $c{'subcountry'};
  407: 				}
  408: 			}
  409: 			$addr{'number'} = $c{'property_identifier'};
  410: 			$addr{'city'}   = $c{'suburb'};
  411: 
  412: 			if (my $rc = $self->_search(\%addr, qw(number road city state country))) {
  413: 				return $self->_cache_and_return($lc, $rc);
  414: 			}
  415: 			if ($addr{'number'}) {
  416: 				if (my $rc = $self->_search(\%addr, qw(road city state country))) {
  417: 					return $self->_cache_and_return($lc, $rc);
  418: 				}
  419: 			}
  420: 
  421: 			# Abort early if no data at all matches this state/country —
  422: 			# saves traversing the full dataset for every remaining strategy.
  423: 			if (!defined($addr{'country'})) {
  424: 				$addr{'country'} = $l =~ /(United States|USA|US)$/i ? 'US'
  425: 					: Carp::croak("TODO: extract country from $l");
  426: 			}
  427: 			# O(1) region check via index built in new() — replaces the former
  428: 			# O(n) grep across all rows.
  429: 			# Major: region_index maps every (state|country) pair in the dataset.
  430: 			# Minor: if target pair is absent, no _search call can match.
  431: 			# Conclusion: early return is equivalent and faster.
  432: 			my $rk = uc($addr{'state'} // '') . '|' . uc($addr{'country'} // '');
  433: 			return $self->_remember_miss($lc) unless $self->{'region_index'}{$rk};
  434: 		}
  435: 	}
  436: 
  437: 	if ($location =~ /^(.+?)[,\s]+(United States|USA|US)$/i) {
  438: 		my $l = $1;
  439: 		$l =~ tr/,/ /;
  440: 		$l =~ s/\s{2,}/ /g;
  441: 
  442: 		# Geo::StreetAddress::US is buggy (RT#122617) — skip county-style addresses
  443: 		if ($location !~ /\sCounty,/i) {
  444: 			my $href = Geo::StreetAddress::US->parse_location($l)
  445: 			        // Geo::StreetAddress::US->parse_address($l);
  446: 
  447: 			if ($href) {

  449: 					$state = _to_two_letter_state('US', $state) if length($state) > 2;
  450: 					my $city   = uc($href->{'city'} // '');
  451: 					if (my $street = $href->{'street'}) {
  452: 						if ($href->{'type'}) {
  453: 							$street .= ' ' . _abbreviate($href->{'type'});
  454: 						}
  455: 						$street .= ' ' . $href->{'suffix'}  if $href->{'suffix'};
  456: 						$street  = $href->{'prefix'} . " $street" if $href->{'prefix'};
  457: 						my %addr = (
  458: 							number  => $href->{'number'},
  459: 							road    => $street,
  460: 							city    => $city,
  461: 							state   => $state,
  462: 							country => 'US',
  463: 						);
  464: 						if ($href->{'number'}) {
  465: 							if (my $rc = $self->_search(\%addr, qw(number road city state country))) {
  466: 								$rc->{'country'} = 'US';
  467: 								return $self->_cache_and_return($lc, $rc);
  468: 							}
  469: 						}
  470: 						if (my $rc = $self->_search(\%addr, qw(road city state country))) {
  471: 							$rc->{'country'} = 'US';
  472: 							return $self->_cache_and_return($lc, $rc);
  473: 						}

  475: 						if ($street && !$href->{'number'}) {
  476: 							$addr{'name'} = delete $addr{'road'};
  477: 							if (my $rc = $self->_search(\%addr, qw(name city state country))) {
  478: 								$rc->{'country'} = 'US';
  479: 								return $self->_cache_and_return($lc, $rc);
  480: 							}
  481: 						}
  482: 					}
  483: 				}
  484: 			}
  485: 		}
  486: 
  487: 		# Fallback: try "name, street, town, state, US" split
  488: 		my @parts = split(/,\s*/, $location);
  489: 		if (scalar(@parts) == 5) {

  492: 				my %addr = (city => $parts[2], state => $state, country => 'US');
  493: 				if ($parts[0] !~ /^\d/) {
  494: 					$addr{'name'} = $parts[0];
  495: 					if ($parts[1] =~ /^(\d+)\s+(.+)/) {
  496: 						$addr{'number'} = $1;
  497: 						$addr{'road'}   = _normalize($2);
  498: 						if (my $rc = $self->_search(\%addr, qw(name number road city state country))) {
  499: 							$rc->{'country'} = 'US';
  500: 							return $self->_cache_and_return($lc, $rc);
  501: 						}
  502: 					} else {
  503: 						$addr{'road'} = _normalize($parts[1]);
  504: 						if (my $rc = $self->_search(\%addr, qw(name road city state country))) {
  505: 							$rc->{'country'} = 'US';
  506: 							return $self->_cache_and_return($lc, $rc);
  507: 						}
  508: 					}
  509: 				} else {
  510: 					$addr{'number'} = $parts[0];
  511: 					$addr{'road'}   = _normalize($parts[1]);
  512: 					if (my $rc = $self->_search(\%addr, qw(number road city state country))) {
  513: 						$rc->{'country'} = 'US';
  514: 						return $self->_cache_and_return($lc, $rc);
  515: 					}
  516: 				}
  517: 			}
  518: 		}
  519: 	}
  520: 
  521: 	# Simple "Town, County, England" — no further sub-parsers will help
  522: 	if (($location =~ /[^,]+,[^,]+,.*England$/) && ($location !~ /[^,]+,[^,]+,[^,]+,.*England$/)) {
  523: 		return $self->_remember_miss($lc);
  524: 	}
  525: 
  526: 	# Geo::Address::Parser — optional; load lazily, skip if not installed
  527: 	unless (Geo::Address::Parser->can('parse')) {
  528: 		eval { require Geo::Address::Parser; Geo::Address::Parser->import() };
  529: 	}
  530: 	if (Geo::Address::Parser->can('parse')) {
  531: 		my $addr_parser = Geo::Address::Parser->new(country => 'UK');
  532: 		if (my $fields = $addr_parser->parse($location)) {
  533: 			# Remove undef fields so _search only matches defined columns
  534: 			delete $fields->{$_} for grep { !defined $fields->{$_} } keys %{$fields};
  535: 			if (my $rc = $self->_search($fields, keys %{$fields})) {
  536: 				$rc->{'country'} = 'UK';
  537: 				return $self->_cache_and_return($lc, $rc);
  538: 			}
  539: 		}
  540: 	}
  541: 
  542: 	# Geo::libpostal — accurate but uses enormous RAM; initialise at most once
  543: 	if ($libpostal_is_installed == LIBPOSTAL_UNKNOWN) {
  544: 		if (eval { require Geo::libpostal; 1 }) {
  545: 			Geo::libpostal->import();
  546: 			$libpostal_is_installed = LIBPOSTAL_INSTALLED;
  547: 		} else {
  548: 			$libpostal_is_installed = LIBPOSTAL_NOT_INSTALLED;
  549: 		}
  550: 	}
  551: 
  552: 	if ($libpostal_is_installed == LIBPOSTAL_INSTALLED) {
  553: 		my %addr = Geo::libpostal::parse_address($location);
  554: 		if (%addr) {
  555: 			# Normalise field names to match our data schema
  556: 			$addr{'number'} = delete $addr{'house_number'} if $addr{'house_number'} && !$addr{'number'};
  557: 			$addr{'name'}   = delete $addr{'house'}        if $addr{'house'}        && !$addr{'name'};
  558: 			$addr{'location'} = $location;
  559: 
  560: 			if (my $street = $addr{'road'}) {
  561: 				$addr{'road'} = _normalize($street);
  562: 			}
  563: 
  564: 			# libpostal returns "england" as a state — map to country code
  565: 			if (defined($addr{'state'}) && !defined($addr{'country'})
  566: 			    && $addr{'state'} eq 'england') {
  567: 				delete $addr{'state'};
  568: 				$addr{'country'} = 'GB';
  569: 			}
  570: 
  571: 			if ($addr{'country'} && ($addr{'state'} || $addr{'state_district'})) {
  572: 				if ($addr{'country'} =~ /Canada/i) {
  573: 					$addr{'country'} = 'Canada';
  574: 					$addr{'state'}   = _to_two_letter_state('Canada', $addr{'state'})
  575: 						if $addr{'state'} && length($addr{'state'}) > 2;
  576: 				} elsif ($addr{'country'} =~ /^(United States|USA|US)$/i) {
  577: 					$addr{'country'} = 'US';
  578: 					$addr{'state'}   = _to_two_letter_state('US', $addr{'state'})
  579: 						if $addr{'state'} && length($addr{'state'}) > 2;
  580: 				}
  581: 
  582: 				if ($addr{'state_district'}) {
  583: 					$addr{'state_district'} =~ s/^(.+)\s+COUNTY\s*$/$1/i;
  584: 					if (my $rc = $self->_search(\%addr, qw(number road city state_district state country))) {
  585: 						return $self->_cache_and_return($lc, $rc);
  586: 					}
  587: 				}
  588: 				if (my $rc = $self->_search(\%addr, qw(number road city state country))) {
  589: 					return $self->_cache_and_return($lc, $rc);
  590: 				}
  591: 				if ($addr{'number'}) {
  592: 					if (my $rc = $self->_search(\%addr, qw(road city state country))) {
  593: 						return $self->_cache_and_return($lc, $rc);
  594: 					}
  595: 				}
  596: 			}
  597: 		}
  598: 	}
  599: 
  600: 	# Last resort: decompose "road, city, state, country" with a regex
  601: 	if ($location =~ /^(.+?),\s*([\s\w]+),\s*([\s\w]+),\s*([\w\s]+)$/) {
  602: 		my %addr = (
  603: 			road    => $1,
  604: 			city    => $2,
  605: 			state   => $3,
  606: 			country => $4,
  607: 		);
  608: 		$addr{'state'}   =~ s/\s+$//g;
  609: 		$addr{'country'} =~ s/\s+$//g;
  610: 
  611: 		if ($addr{'road'} =~ /([\w\s]+),*\s+(.+)/) {
  612: 			$addr{'name'} = $1;
  613: 			$addr{'road'} = $2;
  614: 		}
  615: 		if ($addr{'road'} =~ /^(\d+)\s+(.+)/) {
  616: 			$addr{'number'} = $1;
  617: 			$addr{'road'}   = $2;
  618: 			if (my $rc = $self->_search(\%addr, qw(name number road city state country))) {
  619: 				return $self->_cache_and_return($lc, $rc);
  620: 			}
  621: 		} elsif (my $rc = $self->_search(\%addr, qw(name road city state country))) {
  622: 			return $self->_cache_and_return($lc, $rc);
  623: 		}
  624: 		if ($addr{'name'} && !defined($addr{'number'})) {
  625: 			if (my $rc = $self->_search(\%addr, qw(name road city state country))) {
  626: 				return $self->_cache_and_return($lc, $rc);
  627: 			}
  628: 		}
  629: 	}
  630: 
  631: 	# Try the alternatives table — curated mappings for inconsistent names
  632: 	$location = uc($location);
  633: 	for my $left (keys %alternatives) {
  634: 		next unless $location =~ $left;
  635: 		(my $mapped = $location) =~ s/$left/$alternatives{$left}/;
  636: 		$params{'location'} = $mapped;
  637: 		if (my $rc = $self->geocode(\%params)) {
  638: 			return $self->_cache_and_return($lc, $rc);
  639: 		}
  640: 		if ($mapped =~ /(.+), (England|UK)$/i) {
  641: 			$params{'location'} = "$1, GB";
  642: 			if (my $rc = $self->geocode(\%params)) {
  643: 				return $self->_cache_and_return($lc, $rc);
  644: 			}
  645: 		}
  646: 		$params{'location'} = $location;
  647: 	}
  648: 
  649: 	return $self->_remember_miss($lc);
  650: }

  652: =head2 reverse_geocode
  653: 
  654: =head3 SYNOPSIS
  655: 
  656:     my $loc = $geocoder->reverse_geocode(latlng => '51.3341,-1.4159');
  657:     # Returns the location string(s) for that lat/lon pair.
  658: 
  659: =head3 API SPECIFICATION
  660: 
  661: =head4 input
  662: 
  663:     # Input schema (Params::Validate::Strict) — latlng or lat+lon required
  664:     latlng => { type => 'scalar', optional => 1 }  # "$lat,$long" comma-separated decimal degrees
  665:     lat    => { type => 'scalar', optional => 1 }  # latitude  (alternative to latlng)
  666:     lon    => { type => 'scalar', optional => 1 }  # longitude (alternative to latlng)
  667:     long   => { type => 'scalar', optional => 1 }  # alias for lon
  668: 
  669: =head4 output
  670: 
  671:     # Output schema (Return::Set)
  672:     # scalar context: { type => 'scalar',   optional => 1 }  # location string
  673:     # list context:   { type => 'arrayref', of => { type => 'scalar' } }
  674: 
  675: =cut
  676: 
  677: sub reverse_geocode {
  678: 	my $self = shift;
  679: 
  680: 	if (!ref($self)) {
  681: 		if (scalar @_) {
  682: 			return __PACKAGE__->new()->reverse_geocode(@_);
  683: 		} elsif (!defined($self)) {
  684: 			Carp::croak('Usage: ', __PACKAGE__, '::reverse_geocode(latlng => "$lat,$long")');
  685: 		} elsif ($self eq __PACKAGE__) {
  686: 			Carp::croak("Usage: $self", '::reverse_geocode(latlng => "$lat,$long")');
  687: 		}
  688: 		return __PACKAGE__->new()->reverse_geocode($self);
  689: 	} elsif (ref($self) eq 'HASH') {
  690: 		return __PACKAGE__->new()->reverse_geocode($self);
  691: 	}
  692: 
  693: 	my %params = _normalize_args(
  694: 		'Usage: ' . __PACKAGE__ . '::reverse_geocode(latlng => "$lat,$long")',
  695: 		'latlng', @_
  696: 	);
  697: 
  698: 	my ($latitude, $longitude);
  699: 	if (my $latlng = $params{'latlng'}) {
  700: 		($latitude, $longitude) = split /,/, $latlng;
  701: 	} else {
  702: 		$latitude  = $params{'lat'};
  703: 		$longitude = $params{'lon'} // $params{'long'};
  704: 	}
  705: 
  706: 	if (!defined($latitude) || !defined($longitude)) {
  707: 		Carp::croak('Usage: ', __PACKAGE__, '::reverse_geocode(latlng => "$lat,$long")');
  708: 	}
  709: 
  710: 	my @rc;
  711: 	for my $row (@{$self->{'data'}}) {
  712: 		next unless defined($row->{'latitude'}) && defined($row->{'longitude'});
  713: 		next unless _equal($row->{'latitude'}, $latitude, 4)
  714: 		         && _equal($row->{'longitude'}, $longitude, 4);
  715: 
  716: 		# Rows in $self->{'data'} are blessed as Geo::Location::Point by
  717: 		# the index-building loop in new() — calling as_string() is safe here.
  718: 		my $location = uc($row->as_string());
  719: 		if (wantarray) {
  720: 			push @rc, $location;
  721: 			# Add any reverse-alternative mappings for this location
  722: 			while (my ($left, $right) = each %alternatives) {
  723: 				if ($location =~ $right) {
  724: 					(my $l = $location) =~ s/$right/$left/;
  725: 					push @rc, $l;
  726: 				}
  727: 			}
  728: 		} else {
  729: 			return $location;
  730: 		}
  731: 	}
  732: 	return @rc;
  733: }

  735: =head2 ua
  736: 
  737: Does nothing — present for drop-in compatibility with other Geo::Coder::* modules.
  738: 
  739: =cut
  740: 
  741: ◆sub ua { }  742: ◆  743: ◆# -----------------------------------------------------------------------  744: ◆# Private helpers  745: ◆# -----------------------------------------------------------------------  746: ◆  747: ◆# Purpose:  Normalise the four calling conventions used by geocode/reverse_geocode:  748: ◆#           hashref, even-length key-val list, single bare string.  749: ◆# Entry:    $error_msg  — croak message if an unsupported ref type is passed.  750: ◆#           $bare_key   — hash key to use when a single bare string is given  751: #                         ('location' for geocode, 'latlng' for reverse_geocode).
  752: #           @args       — raw @_ after $self has been shifted.
  753: # Exit:     Flat %params hash.
  754: # Side Effects: None; croaks on non-hash reference arguments.
  755: sub _normalize_args {
  756: 	my ($error_msg, $bare_key, @args) = @_;
  757: 	return %{$args[0]}               if ref($args[0]) eq 'HASH';
  758: 	Carp::croak($error_msg)          if ref($args[0]);
  759: 	return @args                     if @args && @args % 2 == 0;
  760: 	return ($bare_key => $args[0])   if @args == 1;
  761: 	# M7 (logic-gap closure): odd-count > 1 is not a valid calling convention.
  762: 	# Fail fast rather than silently discarding trailing arguments.
  763: 	Carp::croak($error_msg)          if @args;
  764: 	return ();
  765: }
  766: 
  767: # Purpose:  Store $rc in the lookup cache under $key and return it.
  768: #           Reduces the repeated "$self->{cache}{$k} = $r; return $r;" pattern
  769: #           that appeared ~12 times across geocode.
  770: # Entry:    $self, $key (lc string), $rc (Geo::Location::Point or undef).
  771: # Exit:     $rc (returned transparently).
  772: # Side Effects: Writes to $self->{cache}.
  773: sub _cache_and_return {
  774: 	my ($self, $key, $rc) = @_;
  775: 	$self->{'cache'}{$key} = $rc;
  776: 	return $rc;
  777: }
  778: 
  779: # Purpose:  Resolve a full state/province name to its two-letter code using
  780: #           the appropriate Locale:: module.  Returns the original string if
  781: #           no mapping is found or the input is already two characters.
  782: #           Locale::US and Locale::CA objects are cached as module-level
  783: #           singletons to avoid repeated construction overhead.
  784: # Entry:    $country — country name or ISO code; $state — state/province name.
  785: # Exit:     Two-letter code or $state unchanged.
  786: # Side Effects: May initialise $_locale_us or $_locale_ca singletons.
  787: sub _to_two_letter_state {
  788: 	my ($country, $state) = @_;
  789: 	return $state // '' if !defined($state) || length($state) <= 2;
  790: 
  791: 	if ($country =~ /^(United States|USA|US)$/i) {
  792: 		$_locale_us //= Locale::US->new();
  793: 		return $_locale_us->{'state2code'}{uc($state)} // $state;
  794: 	} elsif ($country =~ /Canada/i) {
  795: 		$_locale_ca //= Locale::CA->new();
  796: 		return $_locale_ca->{'province2code'}{uc($state)} // $state;
  797: 	}
  798: 	return $state;
  799: }
  800: 
  801: # Purpose:  Match parsed address components against all data rows.
  802: #           Each named column must match (case-insensitively) the corresponding
  803: #           data field; a minimum of 3 columns must match for a result.
  804: # Entry:    $self, $data — hashref of address components,
  805: #           @columns — ordered list of keys to compare.
  806: # Exit:     Geo::Location::Point on match; undef otherwise.
  807: # Side Effects: May delete undef entries from $data (for undefined optional fields).
  808: sub _search {
  809: 	my ($self, $data, @columns) = @_;
  810: 
  811: 	$self->{'reads'}++;
  812: 
  813: 	# M5 (boolean reduction): the $failed flag was set-and-break in the inner loop
  814: 	# then tested in the outer loop — eliminated by using a labeled 'next ROW'
  815: 	# that short-circuits directly, removing one boolean gate per iteration.
  816: 	ROW: for my $row (@{$self->{'data'}}) {
  817: 		my $matched = 0;
  818: 		my @matched_columns;
  819: 
  820: 		for my $column (@columns) {
  821: 			if (defined($data->{$column})) {
  822: 				next ROW unless defined($row->{$column});
  823: 				next ROW if uc($row->{$column}) ne uc($data->{$column});
  824: 				$matched++;
  825: 				# The accuracy is reported from the columns that matched, not
  826: 				# every column this call site offered: a caller trying
  827: 				# 'number road city...' for a town that has no road must not have
  828: 				# its match described as an address.
  829: 				push @matched_columns, $column;
  830: 			} elsif (exists $data->{$column}) {
  831: 				delete $data->{$column};
  832: 			}
  833: 		}
  834: 
  835: 		next if $matched < 3;
  836: 
  837: 		# Assign confidence based on how many columns contributed to the match
  838: 		my $confidence = $matched == scalar(@columns) ? $CONF_EXACT
  839: 		               : $matched >= 4                ? $CONF_HIGH
  840: 		               :                               $CONF_MEDIUM;
  841: 
  842: 		# Provenance (Phase 6 dependency specs section 2.1).  Set after %{$row}
  843: 		# so that a dataset column that happens to be called accuracy cannot stand
  844: 		# in for the match this code actually made.
  845: 		return Geo::Location::Point->new(
  846: 			location   => $data->{'location'},
  847: 			confidence => $confidence,
  848: 			database   => __PACKAGE__,
  849: 			%{$row},
  850: 			geocoder   => GEOCODER,
  851: 			accuracy   => Geo::Location::Point::accuracy_for(@matched_columns),
  852: 		);
  853: 	}
  854: 	return;
  855: }

  857: =head2 _result_for
  858: 
  859: Build the result for a dataset row that the index matched.  The index holds the
  860: rows themselves - the same hashrefs C<_search> scans - so the object is built
  861: as a copy rather than blessed in place, and the provenance keys every result
  862: carries (section 2.1 of F<BI-Phase6-Dependency-Specs.md>) are added where the
  863: row does not already record them.
  864: 
  865: An index hit is an exact match on the row's string, so the confidence is
  866: C<$CONF_EXACT>, and the accuracy is what the row's own fields justify, through
  867: C<Geo::Location::Point::accuracy_for()>.
  868: 
  869: =cut
  870: 
  871: sub _result_for {
  872: 	my ($self, $row) = @_;
  873: 
  874: 	my %attributes = %{$row};
  875: 	$attributes{'geocoder'}   = GEOCODER;
  876: 	$attributes{'database'}   = __PACKAGE__;
  877: 	$attributes{'confidence'} = $CONF_EXACT;
  878: 	$attributes{'accuracy'}   = Geo::Location::Point::accuracy_for(keys %attributes);
  879: 
  880: 	return Geo::Location::Point->new(\%attributes);
  881: }

  883: # Purpose:  Calculate the geographic centres (centroids) of all city/state/country
  884: #           clusters in the parsed dataset that contain three or more data points.
  885: #           Clusters with fewer than 3 entries are too sparse to be meaningful.
  886: # Entry:    $rows — arrayref of hashrefs with latitude/longitude/city/state/country.
  887: # Exit:     Arrayref of hashrefs (one per qualifying cluster), or undef if none.
  888: # Side Effects: None.
  889: sub _find_geographic_centres {
  890: 	my $rows = $_[0];
  891: 
  892: 	# Group rows by normalised city|state|country key
  893: 	my %groups;
  894: 	for my $row (@{$rows}) {
  895: 		next unless defined($row->{'latitude'})  && defined($row->{'longitude'});
  896: 		next unless $row->{'latitude'}  =~ /^-?\d+\.?\d*$/;
  897: 		next unless $row->{'longitude'} =~ /^-?\d+\.?\d*$/;
  898: 		my $key = join '|',
  899: 			$row->{'city'}    // '',
  900: 			$row->{'state'}   // '',
  901: 			$row->{'country'} // '';
  902: 		push @{$groups{$key}}, $row;
  903: 	}
  904: 
  905: 	my @centres;
  906: 	for my $key (keys %groups) {
  907: 		my $locs = $groups{$key};
  908: 		next if @{$locs} < 3;
  909: 
  910: 		my ($city, $state, $country) = split /\|/, $key;
  911: 		my ($lat, $lon) = _calculate_centre($locs);
  912: 
  913: 		push @centres, {
  914: 			city      => $city,
  915: 			state     => $state,
  916: 			country   => $country,
  917: 			lat       => $lat,
  918: 			latitude  => $lat,
  919: 			longitude => $lon,
  920: 			long      => $lon,
  921: 			lng       => $lon,
  922: 		};
  923: 	}
  924: 
  925: 	return @centres ? \@centres : undef;
  926: }

  928: # Purpose:  Compute the arithmetic mean of latitude and longitude for a group
  929: #           of locations.  Accurate for small geographic areas; for large areas
  930: #           or locations crossing the antimeridian, a vector-mean is needed.
  931: # Entry:    $locs — arrayref of hashrefs each with 'latitude' and 'longitude'.
  932: # Exit:     ($centre_lat, $centre_lon) as decimal degree strings to 6dp.
  933: # Side Effects: None.
  934: sub _calculate_centre {
  935: 	my $locs  = $_[0];
  936: 	my ($sum_lat, $sum_lon) = (0, 0);
  937: 	$sum_lat += $_->{'latitude'}, $sum_lon += $_->{'longitude'} for @{$locs};
  938: 	my $n = scalar @{$locs};
  939: 	return (sprintf('%.6f', $sum_lat / $n), sprintf('%.6f', $sum_lon / $n));
  940: }
  941: 
  942: # https://www.oreilly.com/library/view/perl-cookbook/1565922433/ch02s03.html
  943: # Equal within $dp decimal places — avoids floating-point equality pitfalls.
  944: sub _equal {
  945: 	my ($A, $B, $dp) = @_;
  946: 	return sprintf("%.${dp}g", $A) eq sprintf("%.${dp}g", $B);
  947: }
