   38: =head1 SYNOPSIS
   39: 
   40:     use Geo::Coder::Free::Local;
   41: 
   42:     my $geocoder = Geo::Coder::Free::Local->new();
   43:     my $location = $geocoder->geocode(location => 'Ramsgate, Kent, UK');
   44:     printf "lat=%.6f lon=%.6f\n", $location->lat(), $location->long();
   45: 
   46: =head1 DESCRIPTION
   47: 
   48: Provides geocoding via a user-curated CSV dataset embedded in the module's
   49: C<__DATA__> section.  Locations in the data were verified by GPS and by
   50: inspecting geotagged photographs.  The data is read once at construction time
   51: and indexed for fast lookup.
   52: 
   53: This is the highest-priority backend tried by C<Geo::Coder::Free>.
   54: 
   55: =head1 LIMITATIONS
   56: 
   57: =over 4
   58: 
   59: =item * The embedded C<__DATA__> dataset covers only a small set of hand-picked locations.
   60: There is no mechanism for non-authors to contribute data without patching the module.
   61: 
   62: =item * Canadian and Australian address parsing is not yet implemented; those queries return C<undef>.
   63: 
   64: =item * C<_search> performs an O(n) linear scan through all rows.  The hash-based
   65: index in C<new()> handles exact string matches; no index exists for partial field matches.
   66: 
   67: =item * C<our %alternatives> duplicates mappings in C<Geo::Coder::Free::__DATA__>.
   68: Both should be consolidated into a shared external config file.
   69: 
   70: =item * C<$libpostal_is_installed> is a module-level (effectively global) flag.
   71: Not thread-safe in a forking or threaded Perl deployment.
   72: 
   73: =item * The C<__DATA__> filehandle is a one-shot resource.  The first call to
   74: C<new()> exhausts it with C<my @data = E<lt>DATAE<gt>>; every subsequent
   75: C<new()> in the same process reads zero rows and builds an empty index.
   76: Construct exactly one C<Local> object per process and share it.
   77: 
   78: =item * The hash-index key is C<lc(Geo::Location::Point-E<gt>new($row)-E<gt>as_string())>,
   79: which B<includes the C<name> field>.  To get a direct index hit the caller must
   80: supply the full string with the venue/place name as the leading component;
   81: an address that omits the name falls through to the slower O(n) C<_search> path.
   82: 
   83: =back
   84: 
   85: =cut

  125: =head3 DESCRIPTION
  126: 
  127: Constructor.  Reads the C<__DATA__> CSV block, builds a hash-based lookup
  128: index, and derives the geographic centre of any city/state/country cluster
  129: containing three or more data points.
  130: 
  131: B<One-shot filehandle>: the C<DATA> handle is consumed on the first call to
  132: C<new()>.  Any subsequent call to C<new()> returns an object whose dataset and
  133: index are both empty.  In code that constructs multiple C<Local> objects (e.g.
  134: in tests) construct B<exactly one> instance and reuse it across all callers.
  135: 
  136: =head3 API SPECIFICATION
  137: 
  138: =head4 input
  139: 
  140:     # Input schema (Params::Validate::Strict)
  141:     cache => { type => 'object', optional => 1, can => ['get', 'set'] }  # CHI-compatible cache object
  142: 
  143: =head4 output
  144: 
  145:     # Output schema (Return::Set)
  146:     { type => 'object', isa => 'Geo::Coder::Free::Local' }
  147: 
  148: =head3 FORMAL SPECIFICATION
  149: 
  150:     LocalState ::= ⟨⟨ data  : seq Row;
  151:                        index : Map[STRING → Row];
  152:                        cache : Map[STRING → Point] ⟩⟩
  153: 
  154:     Init : Params → LocalState
  155:     ∀ p : Params •
  156:       let rows    == parse_csv(__DATA__) ∪ geographic_centres(__DATA__) •
  157:       let idx_key == λ r • lc(Point(r).as_string()) •
  158:       LocalState.index = { idx_key(r) ↦ r | r ∈ rows }
  159:

  255: 
  256: =head3 SYNOPSIS
  257: 
  258:     my $pt = $geocoder->geocode(location => '203 E Chatsworth Rd, Reisterstown, Baltimore, MD, US');
  259:     print $pt->lat(), "\n";
  260: 
  261:     # All calling forms are accepted:
  262:     $geocoder->geocode('203 E Chatsworth Rd, Reisterstown, MD, US');
  263:     $geocoder->geocode({ location => '203 E Chatsworth Rd, Reisterstown, MD, US' });
  264: 
  265: =head3 API SPECIFICATION
  266: 
  267: =head4 input
  268: 
  269:     # Input schema (Params::Validate::Strict)
  270:     location => { type => 'scalar' }  # address string; must contain at least two commas
  271: 
  272: =head4 output
  273: 
  274:     # Output schema (Return::Set)
  275:     { type => 'object', isa => 'Geo::Location::Point', optional => 1 }
  276: 
  277: =head3 PROVENANCE
  278: 
  279: Every result carries C<geocoder>, C<database>, C<confidence> and C<accuracy>
  280: (section 2.1 of F<BI-Phase6-Dependency-Specs.md>).  This backend reports
  281: C<database> as C<Geo::Coder::Free::Local>, and C<accuracy> is what the matched
  282: columns justify: a house number makes it an C<address>, a road a C<street>, a
  283: name or city a C<locality>, a state a C<region> and a country a C<country>.  A
  284: lookup answered from the index, which matches a whole row, is an exact match
  285: (C<confidence> of 1.0) whose accuracy comes from the row's own fields - so a
  286: derived geographic centre, which carries a city, state and country and nothing
  287: narrower, is a C<locality>.
  288: 
  289: =head3 MESSAGES
  290: 
  291:     Usage: ...::geocode(...)   No location argument given.
  292: 
  293: =head3 FORMAL SPECIFICATION
  294: 
  295:     Geocode : STRING → Point?
  296:     ∀ addr : STRING •
  297:       let norm == lc(replace_usa(addr)) •
  298:       (norm ∈ cache ⟹ result = cache[norm]) ∧
  299:       (norm ∈ index ⟹ result = index[norm]) ∧
  300:       (¬ result ⟹ result = parse_and_search(addr))
  301: 
  302: =head3 PSEUDOCODE
  303: 
  304:     normalise @_ into %params
  305:     reject if location does not contain two or more commas (not a full address)
  306:     check cache; return hit
  307:     check hash index; return hit and cache it
  308:     # index key includes the 'name' field; supply the venue/place name as the
  309:     # leading address component to guarantee an index hit
  310:     attempt country-specific parser (US, GB; skip CA/AU pending implementation)
  311:     attempt Geo::StreetAddress::US for "..., USA" addresses
  312:     attempt Geo::Address::Parser
  313:     attempt Geo::libpostal (large memory footprint; loaded lazily)
  314:     attempt 4-part regex decomposition (name/road/city/state/country)
  315:     attempt %alternatives mapping
  316:     return undef
  317: 
  318: =cut

  438: 	}

  502: 							$rc->{'country'} = 'US';
  503: 							return $self->_cache_and_return($lc, $rc);
  504: 						}
  505: 					} else {
  506: 						$addr{'road'} = _normalize($parts[1]);
  507: 						if (my $rc = $self->_search(\%addr, qw(name road city state country))) {
  508: 							$rc->{'country'} = 'US';
  509: 							return $self->_cache_and_return($lc, $rc);
  510: 						}
  511: 					}
  512: 				} else {
  513: 					$addr{'number'} = $parts[0];
  514: 					$addr{'road'}   = _normalize($parts[1]);
  515: 					if (my $rc = $self->_search(\%addr, qw(number road city state country))) {
  516: 						$rc->{'country'} = 'US';
  517: 						return $self->_cache_and_return($lc, $rc);
  518: 					}
  519: 				}
  520: 			}
  521: 		}
  522: 	}
  523: 
  524: 	# Simple "Town, County, England" — no further sub-parsers will help
  525: 	if (($location =~ /[^,]+,[^,]+,.*England$/) && ($location !~ /[^,]+,[^,]+,[^,]+,.*England$/)) {
  526: 		return $self->_remember_miss($lc);
  527: 	}
  528: 
  529: 	# Geo::Address::Parser — optional; load lazily, skip if not installed
  530: 	unless (Geo::Address::Parser->can('parse')) {
  531: 		eval { require Geo::Address::Parser; Geo::Address::Parser->import() };
  532: 	}
  533: 	if (Geo::Address::Parser->can('parse')) {
  534: 		my $addr_parser = Geo::Address::Parser->new(country => 'UK');
  535: 		if (my $fields = $addr_parser->parse($location)) {
  536: 			# Remove undef fields so _search only matches defined columns
  537: 			delete $fields->{$_} for grep { !defined $fields->{$_} } keys %{$fields};
  538: 			if (my $rc = $self->_search($fields, keys %{$fields})) {
  539: 				$rc->{'country'} = 'UK';
  540: 				return $self->_cache_and_return($lc, $rc);
  541: 			}
  542: 		}
  543: 	}
  544: 
  545: 	# Geo::libpostal — accurate but uses enormous RAM; initialise at most once
  546: 	if ($libpostal_is_installed == LIBPOSTAL_UNKNOWN) {
  547: 		if (eval { require Geo::libpostal; 1 }) {
  548: 			Geo::libpostal->import();
  549: 			$libpostal_is_installed = LIBPOSTAL_INSTALLED;
  550: 		} else {
  551: 			$libpostal_is_installed = LIBPOSTAL_NOT_INSTALLED;
  552: 		}
  553: 	}
  554: 
  555: 	if ($libpostal_is_installed == LIBPOSTAL_INSTALLED) {
  556: 		my %addr = Geo::libpostal::parse_address($location);
  557: 		if (%addr) {
  558: 			# Normalise field names to match our data schema
  559: 			$addr{'number'} = delete $addr{'house_number'} if $addr{'house_number'} && !$addr{'number'};
  560: 			$addr{'name'}   = delete $addr{'house'}        if $addr{'house'}        && !$addr{'name'};
  561: 			$addr{'location'} = $location;
  562: 
  563: 			if (my $street = $addr{'road'}) {
  564: 				$addr{'road'} = _normalize($street);
  565: 			}
  566: 
  567: 			# libpostal returns "england" as a state — map to country code
  568: 			if (defined($addr{'state'}) && !defined($addr{'country'})
  569: 			    && $addr{'state'} eq 'england') {
  570: 				delete $addr{'state'};
  571: 				$addr{'country'} = 'GB';
  572: 			}
  573: 
  574: 			if ($addr{'country'} && ($addr{'state'} || $addr{'state_district'})) {
  575: 				if ($addr{'country'} =~ /Canada/i) {
  576: 					$addr{'country'} = 'Canada';
  577: 					$addr{'state'}   = _to_two_letter_state('Canada', $addr{'state'})
  578: 						if $addr{'state'} && length($addr{'state'}) > 2;
  579: 				} elsif ($addr{'country'} =~ /^(United States|USA|US)$/i) {
  580: 					$addr{'country'} = 'US';
  581: 					$addr{'state'}   = _to_two_letter_state('US', $addr{'state'})
  582: 						if $addr{'state'} && length($addr{'state'}) > 2;
  583: 				}
  584: 
  585: 				if ($addr{'state_district'}) {
  586: 					$addr{'state_district'} =~ s/^(.+)\s+COUNTY\s*$/$1/i;
  587: 					if (my $rc = $self->_search(\%addr, qw(number road city state_district state country))) {
  588: 						return $self->_cache_and_return($lc, $rc);
  589: 					}
  590: 				}
  591: 				if (my $rc = $self->_search(\%addr, qw(number road city state country))) {
  592: 					return $self->_cache_and_return($lc, $rc);
  593: 				}
  594: 				if ($addr{'number'}) {
  595: 					if (my $rc = $self->_search(\%addr, qw(road city state country))) {
  596: 						return $self->_cache_and_return($lc, $rc);
  597: 					}
  598: 				}
  599: 			}
  600: 		}
  601: 	}
  602: 
  603: 	# Last resort: decompose "road, city, state, country" with a regex
  604: 	if ($location =~ /^(.+?),\s*([\s\w]+),\s*([\s\w]+),\s*([\w\s]+)$/) {
  605: 		my %addr = (
  606: 			road    => $1,
  607: 			city    => $2,
  608: 			state   => $3,
  609: 			country => $4,
  610: 		);
  611: 		$addr{'state'}   =~ s/\s+$//g;
  612: 		$addr{'country'} =~ s/\s+$//g;
  613: 
  614: 		if ($addr{'road'} =~ /([\w\s]+),*\s+(.+)/) {
  615: 			$addr{'name'} = $1;
  616: 			$addr{'road'} = $2;
  617: 		}
  618: 		if ($addr{'road'} =~ /^(\d+)\s+(.+)/) {
  619: 			$addr{'number'} = $1;
  620: 			$addr{'road'}   = $2;
  621: 			if (my $rc = $self->_search(\%addr, qw(name number road city state country))) {
  622: 				return $self->_cache_and_return($lc, $rc);
  623: 			}
  624: 		} elsif (my $rc = $self->_search(\%addr, qw(name road city state country))) {
  625: 			return $self->_cache_and_return($lc, $rc);
  626: 		}
  627: 		if ($addr{'name'} && !defined($addr{'number'})) {
  628: 			if (my $rc = $self->_search(\%addr, qw(name road city state country))) {
  629: 				return $self->_cache_and_return($lc, $rc);
  630: 			}
  631: 		}
  632: 	}
  633: 
  634: 	# Try the alternatives table — curated mappings for inconsistent names
  635: 	$location = uc($location);
  636: 	for my $left (keys %alternatives) {
  637: 		next unless $location =~ $left;
  638: 		(my $mapped = $location) =~ s/$left/$alternatives{$left}/;
  639: 		$params{'location'} = $mapped;
  640: 		if (my $rc = $self->geocode(\%params)) {
  641: 			return $self->_cache_and_return($lc, $rc);
  642: 		}
  643: 		if ($mapped =~ /(.+), (England|UK)$/i) {
  644: 			$params{'location'} = "$1, GB";
  645: 			if (my $rc = $self->geocode(\%params)) {
  646: 				return $self->_cache_and_return($lc, $rc);
  647: 			}
  648: 		}
  649: 		$params{'location'} = $location;
  650: 	}
  651: 
  652: 	return $self->_remember_miss($lc);
  653: }

  655: =head2 reverse_geocode
  656: 
  657: =head3 SYNOPSIS
  658: 
  659:     my $loc = $geocoder->reverse_geocode(latlng => '51.3341,-1.4159');
  660:     # Returns the location string(s) for that lat/lon pair.
  661: 
  662: =head3 API SPECIFICATION
  663: 
  664: =head4 input
  665: 
  666:     # Input schema (Params::Validate::Strict) - latlng or lat+lon required
  667:     latlng => { type => 'scalar', optional => 1 }  # "$lat,$long" comma-separated decimal degrees
  668:     lat    => { type => 'scalar', optional => 1 }  # latitude  (alternative to latlng)
  669:     lon    => { type => 'scalar', optional => 1 }  # longitude (alternative to latlng)
  670:     long   => { type => 'scalar', optional => 1 }  # alias for lon
  671: 
  672: =head4 output
  673: 
  674:     # Output schema (Return::Set)
  675:     # scalar context: { type => 'scalar',   optional => 1 }  # location string
  676:     # list context:   { type => 'arrayref', of => { type => 'scalar' } }
  677: 
  678: =cut
  679: 
  680: sub reverse_geocode {
  681: 	my $self = shift;
  682: 
  683: 	if (!ref($self)) {
  684: 		if (scalar @_) {
  685: 			return __PACKAGE__->new()->reverse_geocode(@_);
  686: 		} elsif (!defined($self)) {
  687: 			Carp::croak('Usage: ', __PACKAGE__, '::reverse_geocode(latlng => "$lat,$long")');
  688: 		} elsif ($self eq __PACKAGE__) {
  689: 			Carp::croak("Usage: $self", '::reverse_geocode(latlng => "$lat,$long")');
  690: 		}
  691: 		return __PACKAGE__->new()->reverse_geocode($self);
  692: 	} elsif (ref($self) eq 'HASH') {
  693: 		return __PACKAGE__->new()->reverse_geocode($self);
  694: 	}
  695: 
  696: 	my %params = _normalize_args(
  697: 		'Usage: ' . __PACKAGE__ . '::reverse_geocode(latlng => "$lat,$long")',
  698: 		'latlng', @_
  699: 	);
  700: 
  701: 	my ($latitude, $longitude);
  702: 	if (my $latlng = $params{'latlng'}) {
  703: 		($latitude, $longitude) = split /,/, $latlng;
  704: 	} else {
  705: 		$latitude  = $params{'lat'};
  706: 		$longitude = $params{'lon'} // $params{'long'};
  707: 	}
  708: 
  709: 	if (!defined($latitude) || !defined($longitude)) {
  710: 		Carp::croak('Usage: ', __PACKAGE__, '::reverse_geocode(latlng => "$lat,$long")');
  711: 	}
  712: 
  713: 	my @rc;
  714: 	for my $row (@{$self->{'data'}}) {
  715: 		next unless defined($row->{'latitude'}) && defined($row->{'longitude'});
  716: 		next unless _equal($row->{'latitude'}, $latitude, 4)
  717: 		         && _equal($row->{'longitude'}, $longitude, 4);
  718: 
  719: 		# Rows in $self->{'data'} are blessed as Geo::Location::Point by
  720: 		# the index-building loop in new() — calling as_string() is safe here.
  721: 		my $location = uc($row->as_string());
  722: 		if (wantarray) {
  723: 			push @rc, $location;
  724: 			# Add any reverse-alternative mappings for this location
  725: 			while (my ($left, $right) = each %alternatives) {
  726: 				if ($location =~ $right) {
  727: 					(my $l = $location) =~ s/$right/$left/;
  728: 					push @rc, $l;
  729: 				}
  730: 			}
  731: 		} else {
  732: 			return $location;
  733: 		}
  734: 	}
  735: 	return @rc;
  736: }

  740: Does nothing - present for drop-in compatibility with other Geo::Coder::* modules.

  744: ◆sub ua { }  745: ◆  746: ◆# -----------------------------------------------------------------------  747: ◆# Private helpers  748: ◆# -----------------------------------------------------------------------  749: ◆  750: ◆# Purpose:  Normalise the four calling conventions used by geocode/reverse_geocode:  751: ◆#           hashref, even-length key-val list, single bare string.  752: ◆# Entry:    $error_msg  — croak message if an unsupported ref type is passed.  753: ◆#           $bare_key   — hash key to use when a single bare string is given  754: #                         ('location' for geocode, 'latlng' for reverse_geocode).
  755: #           @args       — raw @_ after $self has been shifted.
  756: # Exit:     Flat %params hash.
  757: # Side Effects: None; croaks on non-hash reference arguments.
  758: sub _normalize_args {
  759: 	my ($error_msg, $bare_key, @args) = @_;
  760: 	return %{$args[0]}               if ref($args[0]) eq 'HASH';
  761: 	Carp::croak($error_msg)          if ref($args[0]);
  762: 	return @args                     if @args && @args % 2 == 0;
  763: 	return ($bare_key => $args[0])   if @args == 1;
  764: 	# M7 (logic-gap closure): odd-count > 1 is not a valid calling convention.
  765: 	# Fail fast rather than silently discarding trailing arguments.
  766: 	Carp::croak($error_msg)          if @args;
  767: 	return ();
  768: }

  770: # Purpose:  Store $rc in the lookup cache under $key and return it.
  771: #           Reduces the repeated "$self->{cache}{$k} = $r; return $r;" pattern
  772: #           that appeared ~12 times across geocode.
  773: # Entry:    $self, $key (lc string), $rc (Geo::Location::Point or undef).
  774: # Exit:     $rc (returned transparently).
  775: # Side Effects: Writes to $self->{cache}.
  776: sub _cache_and_return {
  777: 	my ($self, $key, $rc) = @_;
  778: 	$self->{'cache'}{$key} = $rc;
  779: 	return $rc;
  780: }

  782: # Purpose:  Resolve a full state/province name to its two-letter code using
  783: #           the appropriate Locale:: module.  Returns the original string if
  784: #           no mapping is found or the input is already two characters.
  785: #           Locale::US and Locale::CA objects are cached as module-level
  786: #           singletons to avoid repeated construction overhead.
  787: # Entry:    $country — country name or ISO code; $state — state/province name.
  788: # Exit:     Two-letter code or $state unchanged.
  789: # Side Effects: May initialise $_locale_us or $_locale_ca singletons.
  790: sub _to_two_letter_state {
  791: 	my ($country, $state) = @_;
  792: 	return $state // '' if !defined($state) || length($state) <= 2;
  793: 
  794: 	if ($country =~ /^(United States|USA|US)$/i) {
  795: 		$_locale_us //= Locale::US->new();
  796: 		return $_locale_us->{'state2code'}{uc($state)} // $state;
  797: 	} elsif ($country =~ /Canada/i) {
  798: 		$_locale_ca //= Locale::CA->new();
  799: 		return $_locale_ca->{'province2code'}{uc($state)} // $state;
  800: 	}
  801: 	return $state;
  802: }

  804: # Purpose:  Match parsed address components against all data rows.
  805: #           Each named column must match (case-insensitively) the corresponding
  806: #           data field; a minimum of 3 columns must match for a result.
  807: # Entry:    $self, $data — hashref of address components,
  808: #           @columns — ordered list of keys to compare.
  809: # Exit:     Geo::Location::Point on match; undef otherwise.
  810: # Side Effects: May delete undef entries from $data (for undefined optional fields).
  811: sub _search {
  812: 	my ($self, $data, @columns) = @_;
  813: 
  814: 	$self->{'reads'}++;
  815: 
  816: 	# M5 (boolean reduction): the $failed flag was set-and-break in the inner loop
  817: 	# then tested in the outer loop — eliminated by using a labeled 'next ROW'
  818: 	# that short-circuits directly, removing one boolean gate per iteration.
  819: 	ROW: for my $row (@{$self->{'data'}}) {
  820: 		my $matched = 0;
  821: 		my @matched_columns;
  822: 
  823: 		for my $column (@columns) {
  824: 			if (defined($data->{$column})) {
  825: 				next ROW unless defined($row->{$column});
  826: 				next ROW if uc($row->{$column}) ne uc($data->{$column});
  827: 				$matched++;
  828: 				# The accuracy is reported from the columns that matched, not
  829: 				# every column this call site offered: a caller trying
  830: 				# 'number road city...' for a town that has no road must not have
  831: 				# its match described as an address.
  832: 				push @matched_columns, $column;
  833: 			} elsif (exists $data->{$column}) {
  834: 				delete $data->{$column};
  835: 			}
  836: 		}
  837: 
  838: 		next if $matched < 3;
  839: 
  840: 		# Assign confidence based on how many columns contributed to the match
  841: 		my $confidence = $matched == scalar(@columns) ? $CONF_EXACT
  842: 		               : $matched >= 4                ? $CONF_HIGH
  843: 		               :                               $CONF_MEDIUM;
  844: 
  845: 		# Provenance (Phase 6 dependency specs section 2.1).  Set after %{$row}
  846: 		# so that a dataset column that happens to be called accuracy cannot stand
  847: 		# in for the match this code actually made.
  848: 		return Geo::Location::Point->new(
  849: 			location   => $data->{'location'},
  850: 			confidence => $confidence,
  851: 			database   => __PACKAGE__,
  852: 			%{$row},
  853: 			geocoder   => GEOCODER,
  854: 			accuracy   => Geo::Location::Point::accuracy_for(@matched_columns),
  855: 		);
  856: 	}
  857: 	return;
  858: }

  860: =head2 _result_for
  861: 
  862: Build the result for a dataset row that the index matched.  The index holds the
  863: rows themselves - the same hashrefs C<_search> scans - so the object is built
  864: as a copy rather than blessed in place, and the provenance keys every result
  865: carries (section 2.1 of F<BI-Phase6-Dependency-Specs.md>) are added where the
  866: row does not already record them.
  867: 
  868: An index hit is an exact match on the row's string, so the confidence is
  869: C<$CONF_EXACT>, and the accuracy is what the row's own fields justify, through
  870: C<Geo::Location::Point::accuracy_for()>.
  871: 
  872: =cut
  873: 
  874: sub _result_for {
  875: 	my ($self, $row) = @_;
  876: 
  877: 	my %attributes = %{$row};
  878: 	$attributes{'geocoder'}   = GEOCODER;
  879: 	$attributes{'database'}   = __PACKAGE__;
  880: 	$attributes{'confidence'} = $CONF_EXACT;
  881: 	$attributes{'accuracy'}   = Geo::Location::Point::accuracy_for(keys %attributes);
  882: 
  883: 	return Geo::Location::Point->new(\%attributes);
  884: }

  886: # Purpose:  Calculate the geographic centres (centroids) of all city/state/country
  887: #           clusters in the parsed dataset that contain three or more data points.
  888: #           Clusters with fewer than 3 entries are too sparse to be meaningful.
  889: # Entry:    $rows — arrayref of hashrefs with latitude/longitude/city/state/country.
  890: # Exit:     Arrayref of hashrefs (one per qualifying cluster), or undef if none.
  891: # Side Effects: None.
  892: sub _find_geographic_centres {
  893: 	my $rows = $_[0];
  894: 
  895: 	# Group rows by normalised city|state|country key
  896: 	my %groups;
  897: 	for my $row (@{$rows}) {
  898: 		next unless defined($row->{'latitude'})  && defined($row->{'longitude'});
  899: 		next unless $row->{'latitude'}  =~ /^-?\d+\.?\d*$/;
  900: 		next unless $row->{'longitude'} =~ /^-?\d+\.?\d*$/;
  901: 		my $key = join '|',
  902: 			$row->{'city'}    // '',
  903: 			$row->{'state'}   // '',
  904: 			$row->{'country'} // '';
  905: 		push @{$groups{$key}}, $row;
  906: 	}
  907: 
  908: 	my @centres;
  909: 	for my $key (keys %groups) {
  910: 		my $locs = $groups{$key};
  911: 		next if @{$locs} < 3;
  912: 
  913: 		my ($city, $state, $country) = split /\|/, $key;
  914: 		my ($lat, $lon) = _calculate_centre($locs);
  915: 
  916: 		push @centres, {
  917: 			city      => $city,
  918: 			state     => $state,
  919: 			country   => $country,
  920: 			lat       => $lat,
  921: 			latitude  => $lat,
  922: 			longitude => $lon,
  923: 			long      => $lon,
  924: 			lng       => $lon,
  925: 		};
  926: 	}
  927: 
  928: 	return @centres ? \@centres : undef;
  929: }

  931: # Purpose:  Compute the arithmetic mean of latitude and longitude for a group
  932: #           of locations.  Accurate for small geographic areas; for large areas
  933: #           or locations crossing the antimeridian, a vector-mean is needed.
  934: # Entry:    $locs — arrayref of hashrefs each with 'latitude' and 'longitude'.
  935: # Exit:     ($centre_lat, $centre_lon) as decimal degree strings to 6dp.
  936: # Side Effects: None.
  937: sub _calculate_centre {
  938: 	my $locs  = $_[0];
  939: 	my ($sum_lat, $sum_lon) = (0, 0);
  940: 	$sum_lat += $_->{'latitude'}, $sum_lon += $_->{'longitude'} for @{$locs};
  941: 	my $n = scalar @{$locs};
  942: 	return (sprintf('%.6f', $sum_lat / $n), sprintf('%.6f', $sum_lon / $n));
  943: }
  944: 
  945: # https://www.oreilly.com/library/view/perl-cookbook/1565922433/ch02s03.html
  946: # Equal within $dp decimal places — avoids floating-point equality pitfalls.
  947: sub _equal {
  948: 	my ($A, $B, $dp) = @_;
  949: 	return sprintf("%.${dp}g", $A) eq sprintf("%.${dp}g", $B);
