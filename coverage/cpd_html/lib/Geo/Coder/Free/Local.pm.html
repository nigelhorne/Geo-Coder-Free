   38: =head1 SYNOPSIS
   39: 
   40:     use Geo::Coder::Free::Local;
   41: 
   42:     my $geocoder = Geo::Coder::Free::Local->new();
   43:     my $location = $geocoder->geocode(location => 'Ramsgate, Kent, UK');
   44:     printf "lat=%.6f lon=%.6f\n", $location->lat(), $location->long();
   45: 
   46: =head1 DESCRIPTION
   47: 
   48: Provides geocoding via a user-curated CSV dataset embedded in the module's
   49: C<__DATA__> section.  Locations in the data were verified by GPS and by
   50: inspecting geotagged photographs.  The data is read once at construction time
   51: and indexed for fast lookup.
   52: 
   53: This is the highest-priority backend tried by C<Geo::Coder::Free>.
   54: 
   55: =head1 LIMITATIONS
   56: 
   57: =over 4
   58: 
   59: =item * The embedded C<__DATA__> dataset covers only a small set of hand-picked locations.
   60: There is no mechanism for non-authors to contribute data without patching the module.
   61: 
   62: =item * Canadian and Australian address parsing is not yet implemented; those queries return C<undef>.
   63: 
   64: =item * C<_search> performs an O(n) linear scan through all rows.  The hash-based
   65: index in C<new()> handles exact string matches; no index exists for partial field matches.
   66: 
   67: =item * C<our %alternatives> duplicates mappings in C<Geo::Coder::Free::__DATA__>.
   68: Both should be consolidated into a shared external config file.
   69: 
   70: =item * C<$libpostal_is_installed> is a module-level (effectively global) flag.
   71: Not thread-safe in a forking or threaded Perl deployment.
   72: 
   73: =item * The C<__DATA__> filehandle is a one-shot resource.  The first call to
   74: C<new()> exhausts it with C<my @data = E<lt>DATAE<gt>>; every subsequent
   75: C<new()> in the same process reads zero rows and builds an empty index.
   76: Construct exactly one C<Local> object per process and share it.
   77: 
   78: =item * The hash-index key is C<lc(Geo::Location::Point-E<gt>new($row)-E<gt>as_string())>,
   79: which B<includes the C<name> field>.  To get a direct index hit the caller must
   80: supply the full string with the venue/place name as the leading component;
   81: an address that omits the name falls through to the slower O(n) C<_search> path.
   82: 
   83: =back
   84: 
   85: =cut

  125: =head3 DESCRIPTION
  126: 
  127: Constructor.  Reads the C<__DATA__> CSV block, builds a hash-based lookup
  128: index, and derives the geographic centre of any city/state/country cluster
  129: containing three or more data points.
  130: 
  131: B<One-shot filehandle>: the C<DATA> handle is consumed on the first call to
  132: C<new()>.  Any subsequent call to C<new()> returns an object whose dataset and
  133: index are both empty.  In code that constructs multiple C<Local> objects (e.g.
  134: in tests) construct B<exactly one> instance and reuse it across all callers.
  135: 
  136: =head3 API SPECIFICATION
  137: 
  138: =head4 input
  139: 
  140:     # Input schema (Params::Validate::Strict)
  141:     cache => { type => 'object', optional => 1, can => ['get', 'set'] }  # CHI-compatible cache object
  142: 
  143: =head4 output
  144: 
  145:     # Output schema (Return::Set)
  146:     { type => 'object', isa => 'Geo::Coder::Free::Local' }
  147: 
  148: =head3 FORMAL SPECIFICATION
  149: 
  150:     LocalState ::= ⟨⟨ data  : seq Row;
  151:                        index : Map[STRING → Row];
  152:                        cache : Map[STRING → Point] ⟩⟩
  153: 
  154:     Init : Params → LocalState
  155:     ∀ p : Params •
  156:       let rows    == parse_csv(__DATA__) ∪ geographic_centres(__DATA__) •
  157:       let idx_key == λ r • lc(Point(r).as_string()) •

  266: =head2 geocode
  267: 
  268: =head3 SYNOPSIS
  269: 
  270:     my $pt = $geocoder->geocode(location => '203 E Chatsworth Rd, Reisterstown, Baltimore, MD, US');
  271:     print $pt->lat(), "\n";
  272: 
  273:     # All calling forms are accepted:
  274:     $geocoder->geocode('203 E Chatsworth Rd, Reisterstown, MD, US');
  275:     $geocoder->geocode({ location => '203 E Chatsworth Rd, Reisterstown, MD, US' });
  276: 
  277: =head3 API SPECIFICATION
  278: 
  279: =head4 input
  280: 
  281:     # Input schema (Params::Validate::Strict)
  282:     location => { type => 'scalar' }  # address string; must contain at least two commas
  283: 
  284: The domain of C<location>:
  285: 
  286: =over 4
  287: 
  288: =item * A whole row of the data matches in any case, and C<USA> is the same as
  289: C<US>.
  290: 
  291: =item * Fewer than two commas, or two commas with nothing between them
  292: (C<'Road,,GB'>), finds nothing and searches nothing.  Two commas with one
  293: character between them (C<'Road, ,GB'>) is enough to go on to the search.
  294: 
  295: =item * Any characters are accepted, and an address that isn't found is remembered
  296: as a miss.  Undecoded bytes are a different key from the characters they encode.
  297: 
  298: =item * C<undef>, C<''> and C<'0'>, an empty hashref or no arguments croak
  299: C<Usage: geocode(location =E<gt> $location)>.  A reference other than a hashref, or an
  300: odd list longer than one, croaks C<Usage: ...::geocode(...)>.
  301: 
  302: =back
  303: 
  304: =head4 output
  305: 
  306:     # Output schema (Return::Set)
  307:     { type => 'object', isa => 'Geo::Location::Point', optional => 1 }
  308: 
  309: =head3 PROVENANCE
  310: 
  311: Every result carries C<geocoder>, C<database>, C<confidence> and C<accuracy>
  312: (section 2.1 of F<BI-Phase6-Dependency-Specs.md>).  This backend reports
  313: C<database> as C<Geo::Coder::Free::Local>, and C<accuracy> is what the matched
  314: columns justify: a house number makes it an C<address>, a road a C<street>, a
  315: name or city a C<locality>, a state a C<region> and a country a C<country>.  A
  316: lookup answered from the index, which matches a whole row, is an exact match
  317: (C<confidence> of 1.0) whose accuracy comes from the row's own fields - so a
  318: derived geographic centre, which carries a city, state and country and nothing
  319: narrower, is a C<locality>.
  320: 
  321: =head3 MESSAGES
  322: 
  323:     Usage: ...::geocode(...)   No location argument given.
  324: 
  325: =head3 FORMAL SPECIFICATION
  326: 
  327:     Geocode : STRING → Point?
  328:     ∀ addr : STRING •
  329:       let norm == lc(replace_usa(addr)) •
  330:       (norm ∈ cache ⟹ result = cache[norm]) ∧
  331:       (norm ∈ index ⟹ result = index[norm]) ∧
  332:       (¬ result ⟹ result = parse_and_search(addr))
  333: 
  334: =head3 PSEUDOCODE
  335: 
  336:     normalise @_ into %params
  337:     reject if location does not contain two or more commas (not a full address)
  338:     check cache; return hit
  339:     check hash index; return hit and cache it
  340:     # index key includes the 'name' field; supply the venue/place name as the
  341:     # leading address component to guarantee an index hit
  342:     attempt country-specific parser (US, GB; skip CA/AU pending implementation)
  343:     attempt Geo::StreetAddress::US for "..., USA" addresses
  344:     attempt Geo::Address::Parser
  345:     attempt Geo::libpostal (large memory footprint; loaded lazily)
  346:     attempt 4-part regex decomposition (name/road/city/state/country)
  347:     attempt %alternatives mapping
  348:     return undef
  349: 
  350: =cut

  353: 	my $self = shift;
  354: 
  355: 	# Handle being called as a function rather than a method
  356: 	if (!ref($self)) {
  357: 		if (scalar @_) {
  358: 			return __PACKAGE__->new()->geocode(@_);
  359: 		} elsif (!defined($self)) {
  360: 			Carp::croak('Usage: ', __PACKAGE__, '::geocode(location => $location)');
  361: 		} elsif ($self eq __PACKAGE__) {
  362: 			Carp::croak("Usage: $self", '::geocode(location => $location)');
  363: 		}
  364: 		return __PACKAGE__->new()->geocode($self);
  365: 	} elsif (ref($self) eq 'HASH') {
  366: 		return __PACKAGE__->new()->geocode($self);
  367: 	}
  368: 
  369: 	my %params = _normalize_args(
  370: 		'Usage: ' . __PACKAGE__ . '::geocode(location => $location)',
  371: 		'location', @_
  372: 	);
  373: 
  374: 	my $location = $params{'location'}
  375: 		or Carp::croak('Usage: geocode(location => $location)');
  376: 
  377: 	# Normalise the lookup key - "USA" is treated the same as "US"
  378: 	my $lc = lc($location);
  379: 	$lc =~ s/,\s*usa$/, us/i;
  380: 
  381: 	return $self->{'cache'}{$lc}  if exists $self->{'cache'}{$lc};
  382: 	return $self->_cache_and_return($lc, $self->_result_for($self->{'index'}{$lc}))
  383: 		if exists $self->{'index'}{$lc};
  384: 
  385: 	# A miss found by an earlier instance, or an earlier call through this
  386: 	# one, means the same search would find nothing again: the data does not
  387: 	# change while the process runs.  clear_miss_cache() is for when it has.
  388: 	return if $self->_missed($lc);
  389: 
  390: 	# This backend only handles full addresses (at least "road, city, country")
  391: 	return $self->_remember_miss($lc) if $location !~ /,.+,/;

  396: 			country => 'US', auto_clean => 1, force_case => 1, force_post_code => 0
  397: 		);
  398: 	} elsif ($location =~ /(England|Scotland|Wales|Northern Ireland|UK|GB)$/i) {
  399: 		$ap = $self->{'ap'}{'gb'} //= Lingua::EN::AddressParse->new(
  400: 			country => 'GB', auto_clean => 1, force_case => 1, force_post_code => 0
  401: 		);
  402: 	} elsif ($location =~ /Canada$/) {
  403: 		# TODO: Canadian address parsing not yet implemented
  404: 		return $self->_remember_miss($lc);
  405: 	} elsif ($location =~ /Australia$/) {
  406: 		# TODO: Australian address parsing not yet implemented
  407: 		return $self->_remember_miss($lc);
  408: 	}
  409: 
  410: 	if ($ap) {
  411: 		my $l = $location;
  412: 		$l =~ s/(.+), (England|UK)$/$1, GB/i;
  413: 
  414: 		if ($ap->parse($l) == 0) {
  415: 			my %c = $ap->components();
  416: 			my %addr = (location => $l);
  417: 
  418: 			if (my $street = $c{'street_name'}) {
  419: 				if (my $type = $c{'street_type'}) {
  420: 					my $abbrev = _abbreviate($type);
  421: 					$street .= ' ' . ($abbrev || $type);
  422: 					$street .= ' ' . $c{'street_direction_suffix'}

  424: 					$street =~ s/^0+//;
  425: 					$addr{'road'} = $street;
  426: 				}
  427: 			}

  430: 				$addr{'state'} = $c{'subcountry'};
  431: 			} else {
  432: 				my $country_raw = $c{'country'} // '';
  433: 				if ($country_raw =~ /Canada/i) {
  434: 					$addr{'country'} = 'CA';
  435: 					$addr{'state'}   = _to_two_letter_state('Canada', $c{'subcountry'});

  437: 					$addr{'country'} = 'US';
  438: 					$addr{'state'}   = _to_two_letter_state('US', $c{'subcountry'});
  439: 				} elsif ($country_raw) {
  440: 					$addr{'country'} = $country_raw;
  441: 					$addr{'state'}   = $c{'subcountry'} if $c{'subcountry'};
  442: 				}
  443: 			}
  444: 			$addr{'number'} = $c{'property_identifier'};
  445: 			$addr{'city'}   = $c{'suburb'};
  446: 
  447: 			if (my $rc = $self->_search(\%addr, qw(number road city state country))) {
  448: 				return $self->_cache_and_return($lc, $rc);
  449: 			}
  450: 			if ($addr{'number'}) {
  451: 				if (my $rc = $self->_search(\%addr, qw(road city state country))) {
  452: 					return $self->_cache_and_return($lc, $rc);
  453: 				}
  454: 			}
  455: 
  456: 			# Abort early if no data at all matches this state/country -
  457: 			# saves traversing the full dataset for every remaining strategy.

  460: 				return $self->_remember_miss($lc) if $l !~ /(United States|USA|US)$/i;
  461: 				$addr{'country'} = 'US';
  462: 			}

  464: 			# O(n) grep across all rows.
  465: 			# Major: region_index maps every (state|country) pair in the dataset.
  466: 			# Minor: if target pair is absent, no _search call can match.
  467: 			# Conclusion: early return is equivalent and faster.

  469: 			return $self->_remember_miss($lc) unless $self->{'region_index'}{$rk};
  470: 		}
  471: 	}
  472: 
  473: 	if ($location =~ /^(.+?)[,\s]+(United States|USA|US)$/i) {
  474: 		my $l = $1;

  688: =head2 reverse_geocode
  689: 
  690: =head3 SYNOPSIS
  691: 
  692:     my $loc = $geocoder->reverse_geocode(latlng => '51.3341,-1.4159');
  693:     # Returns the location string(s) for that lat/lon pair.
  694: 
  695: =head3 API SPECIFICATION
  696: 
  697: =head4 input
  698: 
  699:     # Input schema (Params::Validate::Strict) - latlng or lat+lon required
  700:     latlng => { type => 'scalar', optional => 1 }  # "$lat,$long" comma-separated decimal degrees
  701:     lat    => { type => 'scalar', optional => 1 }  # latitude  (alternative to latlng)
  702:     lon    => { type => 'scalar', optional => 1 }  # longitude (alternative to latlng)
  703:     long   => { type => 'scalar', optional => 1 }  # alias for lon
  704: 
  705: The domain of the coordinates:
  706: 
  707: =over 4
  708: 
  709: =item * Each is compared to the data at four significant figures, not four decimal
  710: places: C<39.4749> and C<39.47> both match a row at C<39.46727>, and C<39.4751>
  711: does not.
  712: 
  713: =item * No range check applies.  C<'0,0'> and C<'91,181'> find nothing and are
  714: not errors.  A third part is dropped.
  715: 
  716: =item * Each must be an optional minus, ASCII digits, then an optional point and more
  717: digits, as in L<Geo::Coder::Free::MaxMind/reverse_geocode>.  Anything else croaks
  718: C<Invalid coordinate value for reverse_geocode>.
  719: 
  720: =item * C<latlng> is used when it is true.  No arguments, C<''> or C<'0'>, one part
  721: with no comma, or only one of C<lat> and C<lon>, croak C<Usage:>.  So do a reference
  722: other than a hashref and an odd list longer than one.
  723: 
  724: =item * Nothing found is C<0> in scalar context and an empty list in list context.
  725: 
  726: =back
  727: 
  728: =head4 output
  729: 
  730:     # Output schema (Return::Set)
  731:     # scalar context: { type => 'scalar',   optional => 1 }  # location string
  732:     # list context:   { type => 'arrayref', of => { type => 'scalar' } }
  733: 
  734: =cut

  736: sub reverse_geocode {
  737: 	my $self = shift;
  738: 
  739: 	if (!ref($self)) {
  740: 		if (scalar @_) {
  741: 			return __PACKAGE__->new()->reverse_geocode(@_);
  742: 		} elsif (!defined($self)) {
  743: 			Carp::croak('Usage: ', __PACKAGE__, '::reverse_geocode(latlng => "$lat,$long")');
  744: 		} elsif ($self eq __PACKAGE__) {
  745: 			Carp::croak("Usage: $self", '::reverse_geocode(latlng => "$lat,$long")');
  746: 		}
  747: 		return __PACKAGE__->new()->reverse_geocode($self);
  748: 	} elsif (ref($self) eq 'HASH') {
  749: 		return __PACKAGE__->new()->reverse_geocode($self);
  750: 	}
  751: 
  752: 	my %params = _normalize_args(
  753: 		'Usage: ' . __PACKAGE__ . '::reverse_geocode(latlng => "$lat,$long")',
  754: 		'latlng', @_
  755: 	);
  756: 
  757: 	my ($latitude, $longitude);
  758: 	if (my $latlng = $params{'latlng'}) {
  759: 		($latitude, $longitude) = split /,/, $latlng;
  760: 	} else {
  761: 		$latitude  = $params{'lat'};
  762: 		$longitude = $params{'lon'} // $params{'long'};
  763: 	}
  764: 
  765: 	if (!defined($latitude) || !defined($longitude)) {
  766: 		Carp::croak('Usage: ', __PACKAGE__, '::reverse_geocode(latlng => "$lat,$long")');
  767: 	}
  768: 
  769: 	# The same test MaxMind::reverse_geocode applies, so the backends agree
  770: 	for my $c ($latitude, $longitude) {
  771: 		Carp::croak('Invalid coordinate value for reverse_geocode')
  772: 			unless $c =~ /\A-?[0-9]+\.?[0-9]*\z/;
  773: 	}
  774: 
  775: 	my @rc;
  776: 	for my $row (@{$self->{'data'}}) {
  777: 		next unless defined($row->{'latitude'}) && defined($row->{'longitude'});
  778: 		next unless _equal($row->{'latitude'}, $latitude, 4)
  779: 		         && _equal($row->{'longitude'}, $longitude, 4);

  781: 		# Rows in $self->{'data'} are blessed as Geo::Location::Point by
  782: 		# the index-building loop in new() - calling as_string() is safe here.
  783: 		my $location = uc($row->as_string());
  784: 		if (wantarray) {
  785: 			push @rc, $location;
  786: 			# Add any reverse-alternative mappings for this location
  787: 			# keys, not each(): %alternatives is package-wide, and an each()
  788: 			# iterator left part-way would make the next pass skip entries
  789: 			foreach my $left (sort keys %alternatives) {
  790: 				my $right = $alternatives{$left};
  791: 				if ($location =~ $right) {
  792: 					(my $l = $location) =~ s/$right/$left/;
  793: 					push @rc, $l;
  794: 				}
  795: 			}
  796: 		} else {
  797: 			return $location;
  798: 		}
  799: 	}
  800: 	return @rc;
  801: }

  881: # Purpose:  Match parsed address components against all data rows.
  882: #           Each named column must match (case-insensitively) the corresponding
  883: #           data field; a minimum of 3 columns must match for a result.
  884: # Entry:    $self, $data - hashref of address components,
  885: #           @columns - ordered list of keys to compare.

  888: sub _search {
  889: 	my ($self, $data, @columns) = @_;
  890: 
  891: 	$self->{'reads'}++;
  892: 
  893: 	# M5 (boolean reduction): the $failed flag was set-and-break in the inner loop
  894: 	# then tested in the outer loop - eliminated by using a labeled 'next ROW'
  895: 	# that short-circuits directly, removing one boolean gate per iteration.
  896: 	ROW: for my $row (@{$self->{'data'}}) {
  897: 		my $matched = 0;
  898: 		my @matched_columns;
  899: 
  900: 		for my $column (@columns) {
  901: 			if (defined($data->{$column})) {
  902: 				next ROW unless defined($row->{$column});
  903: 				next ROW if uc($row->{$column}) ne uc($data->{$column});
  904: 				$matched++;
  905: 				# The accuracy is reported from the columns that matched, not
  906: 				# every column this call site offered: a caller trying
  907: 				# 'number road city...' for a town that has no road must not have
  908: 				# its match described as an address.
  909: 				push @matched_columns, $column;
  910: 			} elsif (exists $data->{$column}) {
  911: 				delete $data->{$column};
  912: 			}
  913: 		}
  914: 
  915: 		next if $matched < 3;
  916: 
  917: 		# Assign confidence based on how many columns contributed to the match
  918: 		my $confidence = $matched == scalar(@columns) ? $CONF_EXACT
  919: 		               : $matched >= 4                ? $CONF_HIGH
  920: 		               :                               $CONF_MEDIUM;
  921: 
  922: 		# Provenance (Phase 6 dependency specs section 2.1).  Set after %{$row}
  923: 		# so that a dataset column that happens to be called accuracy cannot stand
  924: 		# in for the match this code actually made.
  925: 		return Geo::Location::Point->new(
  926: 			location   => $data->{'location'},
  927: 			confidence => $confidence,
  928: 			database   => __PACKAGE__,
  929: 			%{$row},
  930: 			geocoder   => GEOCODER,
  931: 			accuracy   => Geo::Location::Point::accuracy_for(@matched_columns),
  932: 		);
  933: 	}
  934: 	return;
  935: }

  940: rows themselves - the same hashrefs C<_search> scans - so the object is built

  988: 		next if @{$locs} < 3;
  989: 
  990: 		my ($city, $state, $country) = split /\|/, $key;
  991: 		my ($lat, $lon) = _calculate_centre($locs);
  992: 
  993: 		push @centres, {
  994: 			city      => $city,
  995: 			state     => $state,
  996: 			country   => $country,
  997: 			lat       => $lat,
  998: 			latitude  => $lat,
  999: 			longitude => $lon,
 1000: 			long      => $lon,
 1001: 			lng       => $lon,
 1002: 		};
 1003: 	}
 1004: 
 1005: 	return @centres ? \@centres : undef;
 1006: }
 1007: 
 1008: # Purpose:  Compute the arithmetic mean of latitude and longitude for a group
 1009: #           of locations.  Accurate for small geographic areas; for large areas
 1010: #           or locations crossing the antimeridian, a vector-mean is needed.
 1011: # Entry:    $locs - arrayref of hashrefs each with 'latitude' and 'longitude'.
 1012: # Exit:     ($centre_lat, $centre_lon) as decimal degree strings to 6dp.
 1013: # Side Effects: None.
 1014: sub _calculate_centre {
 1015: 	my $locs  = $_[0];
 1016: 	my ($sum_lat, $sum_lon) = (0, 0);
 1017: 	$sum_lat += $_->{'latitude'}, $sum_lon += $_->{'longitude'} for @{$locs};
 1018: 	my $n = scalar @{$locs};
 1019: 	return (sprintf('%.6f', $sum_lat / $n), sprintf('%.6f', $sum_lon / $n));
 1020: }
 1021: 
 1022: # https://www.oreilly.com/library/view/perl-cookbook/1565922433/ch02s03.html
 1023: # Equal within $dp decimal places - avoids floating-point equality pitfalls.
 1024: sub _equal {
 1025: 	my ($A, $B, $dp) = @_;
 1026: 	return sprintf("%.${dp}g", $A) eq sprintf("%.${dp}g", $B);
 1027: }
