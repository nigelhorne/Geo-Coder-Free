   38: =head1 SYNOPSIS
   39: 
   40:     use Geo::Coder::Free::Local;
   41: 
   42:     my $geocoder = Geo::Coder::Free::Local->new();
   43:     my $location = $geocoder->geocode(location => 'Ramsgate, Kent, UK');
   44:     printf "lat=%.6f lon=%.6f\n", $location->lat(), $location->long();
   45: 
   46: =head1 DESCRIPTION
   47: 
   48: Provides geocoding via a user-curated CSV dataset embedded in the module's
   49: C<__DATA__> section.  Locations in the data were verified by GPS and by
   50: inspecting geotagged photographs.  The data is read once at construction time
   51: and indexed for fast lookup.
   52: 
   53: This is the highest-priority backend tried by C<Geo::Coder::Free>.
   54: 
   55: =head1 LIMITATIONS
   56: 
   57: =over 4
   58: 
   59: =item * The embedded C<__DATA__> dataset covers only a small set of hand-picked locations.
   60: There is no mechanism for non-authors to contribute data without patching the module.
   61: 
   62: =item * Canadian and Australian address parsing is not yet implemented; those queries return C<undef>.
   63: 
   64: =item * C<_search> performs an O(n) linear scan through all rows.  The hash-based
   65: index in C<new()> handles exact string matches; no index exists for partial field matches.
   66: 
   67: =item * C<our %alternatives> duplicates mappings in C<Geo::Coder::Free::__DATA__>.
   68: Both should be consolidated into a shared external config file.
   69: 
   70: =item * C<$libpostal_is_installed> is a module-level (effectively global) flag.
   71: Not thread-safe in a forking or threaded Perl deployment.
   72: 
   73: =item * The C<__DATA__> filehandle is a one-shot resource.  The first call to
   74: C<new()> exhausts it with C<my @data = E<lt>DATAE<gt>>; every subsequent
   75: C<new()> in the same process reads zero rows and builds an empty index.
   76: Construct exactly one C<Local> object per process and share it.
   77: 
   78: =item * The hash-index key is C<lc(Geo::Location::Point-E<gt>new($row)-E<gt>as_string())>,
   79: which B<includes the C<name> field>.  To get a direct index hit the caller must
   80: supply the full string with the venue/place name as the leading component;
   81: an address that omits the name falls through to the slower O(n) C<_search> path.
   82: 
   83: =back
   84: 
   85: =cut

  125: =head3 DESCRIPTION
  126: 
  127: Constructor.  Reads the C<__DATA__> CSV block, builds a hash-based lookup
  128: index, and derives the geographic centre of any city/state/country cluster
  129: containing three or more data points.
  130: 
  131: B<One-shot filehandle>: the C<DATA> handle is consumed on the first call to
  132: C<new()>.  Any subsequent call to C<new()> returns an object whose dataset and
  133: index are both empty.  In code that constructs multiple C<Local> objects (e.g.
  134: in tests) construct B<exactly one> instance and reuse it across all callers.
  135: 
  136: =head3 API SPECIFICATION
  137: 
  138: =head4 input
  139: 
  140:     # Input schema (Params::Validate::Strict)
  141:     cache => { type => 'object', optional => 1, can => ['get', 'set'] }  # CHI-compatible cache object
  142: 
  143: =head4 output
  144: 
  145:     # Output schema (Return::Set)
  146:     { type => 'object', isa => 'Geo::Coder::Free::Local' }
  147: 
  148: =head3 FORMAL SPECIFICATION
  149: 
  150:     LocalState ::= ⟨⟨ data  : seq Row;
  151:                        index : Map[STRING → Row];
  152:                        cache : Map[STRING → Point] ⟩⟩
  153: 
  154:     Init : Params → LocalState
  155:     ∀ p : Params •
  156:       let rows    == parse_csv(__DATA__) ∪ geographic_centres(__DATA__) •
  157:       let idx_key == λ r • lc(Point(r).as_string()) •
  158:       LocalState.index = { idx_key(r) ↦ r | r ∈ rows }

  267: 
  268: =head3 SYNOPSIS
  269: 
  270:     my $pt = $geocoder->geocode(location => '203 E Chatsworth Rd, Reisterstown, Baltimore, MD, US');
  271:     print $pt->lat(), "\n";
  272: 
  273:     # All calling forms are accepted:
  274:     $geocoder->geocode('203 E Chatsworth Rd, Reisterstown, MD, US');
  275:     $geocoder->geocode({ location => '203 E Chatsworth Rd, Reisterstown, MD, US' });
  276: 
  277: =head3 API SPECIFICATION
  278: 
  279: =head4 input
  280: 
  281:     # Input schema (Params::Validate::Strict)
  282:     location => { type => 'scalar' }  # address string; must contain at least two commas
  283: 
  284: =head4 output
  285: 
  286:     # Output schema (Return::Set)
  287:     { type => 'object', isa => 'Geo::Location::Point', optional => 1 }
  288: 
  289: =head3 PROVENANCE
  290: 
  291: Every result carries C<geocoder>, C<database>, C<confidence> and C<accuracy>
  292: (section 2.1 of F<BI-Phase6-Dependency-Specs.md>).  This backend reports
  293: C<database> as C<Geo::Coder::Free::Local>, and C<accuracy> is what the matched
  294: columns justify: a house number makes it an C<address>, a road a C<street>, a
  295: name or city a C<locality>, a state a C<region> and a country a C<country>.  A
  296: lookup answered from the index, which matches a whole row, is an exact match
  297: (C<confidence> of 1.0) whose accuracy comes from the row's own fields - so a
  298: derived geographic centre, which carries a city, state and country and nothing
  299: narrower, is a C<locality>.
  300: 
  301: =head3 MESSAGES
  302: 
  303:     Usage: ...::geocode(...)   No location argument given.
  304: 
  305: =head3 FORMAL SPECIFICATION
  306: 
  307:     Geocode : STRING → Point?
  308:     ∀ addr : STRING •
  309:       let norm == lc(replace_usa(addr)) •
  310:       (norm ∈ cache ⟹ result = cache[norm]) ∧
  311:       (norm ∈ index ⟹ result = index[norm]) ∧
  312:       (¬ result ⟹ result = parse_and_search(addr))
  313: 
  314: =head3 PSEUDOCODE
  315: 
  316:     normalise @_ into %params
  317:     reject if location does not contain two or more commas (not a full address)
  318:     check cache; return hit
  319:     check hash index; return hit and cache it
  320:     # index key includes the 'name' field; supply the venue/place name as the
  321:     # leading address component to guarantee an index hit
  322:     attempt country-specific parser (US, GB; skip CA/AU pending implementation)
  323:     attempt Geo::StreetAddress::US for "..., USA" addresses
  324:     attempt Geo::Address::Parser
  325:     attempt Geo::libpostal (large memory footprint; loaded lazily)
  326:     attempt 4-part regex decomposition (name/road/city/state/country)
  327:     attempt %alternatives mapping
  328:     return undef
  329: 
  330: =cut

  355: 		or Carp::croak('Usage: geocode(location => $location)');

  363: 		if exists $self->{'index'}{$lc};

  517: 					} else {
  518: 						$addr{'road'} = _normalize($parts[1]);
  519: 						if (my $rc = $self->_search(\%addr, qw(name road city state country))) {
  520: 							$rc->{'country'} = 'US';
  521: 							return $self->_cache_and_return($lc, $rc);
  522: 						}
  523: 					}
  524: 				} else {
  525: 					$addr{'number'} = $parts[0];
  526: 					$addr{'road'}   = _normalize($parts[1]);
  527: 					if (my $rc = $self->_search(\%addr, qw(number road city state country))) {
  528: 						$rc->{'country'} = 'US';
  529: 						return $self->_cache_and_return($lc, $rc);
  530: 					}
  531: 				}
  532: 			}
  533: 		}
  534: 	}

  536: 	# Simple "Town, County, England" — no further sub-parsers will help
  537: 	if (($location =~ /[^,]+,[^,]+,.*England$/) && ($location !~ /[^,]+,[^,]+,[^,]+,.*England$/)) {
  538: 		return $self->_remember_miss($lc);

  542: 	unless (Geo::Address::Parser->can('parse')) {

  600: 						return $self->_cache_and_return($lc, $rc);
  601: 					}
  602: 				}
  603: 				if (my $rc = $self->_search(\%addr, qw(number road city state country))) {
  604: 					return $self->_cache_and_return($lc, $rc);
  605: 				}
  606: 				if ($addr{'number'}) {
  607: 					if (my $rc = $self->_search(\%addr, qw(road city state country))) {
  608: 						return $self->_cache_and_return($lc, $rc);
  609: 					}
  610: 				}
  611: 			}
  612: 		}
  613: 	}
  614: 
  615: 	# Last resort: decompose "road, city, state, country" with a regex
  616: 	if ($location =~ /^(.+?),\s*([\s\w]+),\s*([\s\w]+),\s*([\w\s]+)$/) {
  617: 		my %addr = (
  618: 			road    => $1,
  619: 			city    => $2,
  620: 			state   => $3,
  621: 			country => $4,
  622: 		);
  623: 		$addr{'state'}   =~ s/\s+$//g;
  624: 		$addr{'country'} =~ s/\s+$//g;

  626: 		if ($addr{'road'} =~ /([\w\s]+),*\s+(.+)/) {
  627: 			$addr{'name'} = $1;
  628: 			$addr{'road'} = $2;
  629: 		}
  630: 		if ($addr{'road'} =~ /^(\d+)\s+(.+)/) {
  631: 			$addr{'number'} = $1;
  632: 			$addr{'road'}   = $2;
  633: 			if (my $rc = $self->_search(\%addr, qw(name number road city state country))) {
  634: 				return $self->_cache_and_return($lc, $rc);
  635: 			}
  636: 		} elsif (my $rc = $self->_search(\%addr, qw(name road city state country))) {
  637: 			return $self->_cache_and_return($lc, $rc);
  638: 		}
  639: 		if ($addr{'name'} && !defined($addr{'number'})) {
  640: 			if (my $rc = $self->_search(\%addr, qw(name road city state country))) {
  641: 				return $self->_cache_and_return($lc, $rc);
  642: 			}
  643: 		}
  644: 	}
  645: 
  646: 	# Try the alternatives table — curated mappings for inconsistent names
  647: 	$location = uc($location);
  648: 	for my $left (keys %alternatives) {
  649: 		next unless $location =~ $left;
  650: 		(my $mapped = $location) =~ s/$left/$alternatives{$left}/;
  651: 		$params{'location'} = $mapped;
  652: 		if (my $rc = $self->geocode(\%params)) {
  653: 			return $self->_cache_and_return($lc, $rc);
  654: 		}
  655: 		if ($mapped =~ /(.+), (England|UK)$/i) {
  656: 			$params{'location'} = "$1, GB";
  657: 			if (my $rc = $self->geocode(\%params)) {
  658: 				return $self->_cache_and_return($lc, $rc);
  659: 			}
  660: 		}
  661: 		$params{'location'} = $location;
  662: 	}
  663: 
  664: 	return $self->_remember_miss($lc);
  665: }

  667: =head2 reverse_geocode
  668: 
  669: =head3 SYNOPSIS
  670: 
  671:     my $loc = $geocoder->reverse_geocode(latlng => '51.3341,-1.4159');
  672:     # Returns the location string(s) for that lat/lon pair.
  673: 
  674: =head3 API SPECIFICATION
  675: 
  676: =head4 input
  677: 
  678:     # Input schema (Params::Validate::Strict) - latlng or lat+lon required
  679:     latlng => { type => 'scalar', optional => 1 }  # "$lat,$long" comma-separated decimal degrees
  680:     lat    => { type => 'scalar', optional => 1 }  # latitude  (alternative to latlng)
  681:     lon    => { type => 'scalar', optional => 1 }  # longitude (alternative to latlng)
  682:     long   => { type => 'scalar', optional => 1 }  # alias for lon
  683: 
  684: =head4 output
  685: 
  686:     # Output schema (Return::Set)
  687:     # scalar context: { type => 'scalar',   optional => 1 }  # location string
  688:     # list context:   { type => 'arrayref', of => { type => 'scalar' } }
  689: 
  690: =cut
  691: 
  692: sub reverse_geocode {
  693: 	my $self = shift;
  694: 
  695: 	if (!ref($self)) {
  696: 		if (scalar @_) {
  697: 			return __PACKAGE__->new()->reverse_geocode(@_);
  698: 		} elsif (!defined($self)) {
  699: 			Carp::croak('Usage: ', __PACKAGE__, '::reverse_geocode(latlng => "$lat,$long")');
  700: 		} elsif ($self eq __PACKAGE__) {
  701: 			Carp::croak("Usage: $self", '::reverse_geocode(latlng => "$lat,$long")');
  702: 		}
  703: 		return __PACKAGE__->new()->reverse_geocode($self);
  704: 	} elsif (ref($self) eq 'HASH') {
  705: 		return __PACKAGE__->new()->reverse_geocode($self);
  706: 	}
  707: 
  708: 	my %params = _normalize_args(
  709: 		'Usage: ' . __PACKAGE__ . '::reverse_geocode(latlng => "$lat,$long")',
  710: 		'latlng', @_
  711: 	);
  712: 
  713: 	my ($latitude, $longitude);
  714: 	if (my $latlng = $params{'latlng'}) {
  715: 		($latitude, $longitude) = split /,/, $latlng;
  716: 	} else {
  717: 		$latitude  = $params{'lat'};
  718: 		$longitude = $params{'lon'} // $params{'long'};
  719: 	}
  720: 
  721: 	if (!defined($latitude) || !defined($longitude)) {
  722: 		Carp::croak('Usage: ', __PACKAGE__, '::reverse_geocode(latlng => "$lat,$long")');
  723: 	}
  724: 
  725: 	my @rc;
  726: 	for my $row (@{$self->{'data'}}) {
  727: 		next unless defined($row->{'latitude'}) && defined($row->{'longitude'});
  728: 		next unless _equal($row->{'latitude'}, $latitude, 4)
  729: 		         && _equal($row->{'longitude'}, $longitude, 4);
  730: 
  731: 		# Rows in $self->{'data'} are blessed as Geo::Location::Point by
  732: 		# the index-building loop in new() — calling as_string() is safe here.
  733: 		my $location = uc($row->as_string());
  734: 		if (wantarray) {
  735: 			push @rc, $location;
  736: 			# Add any reverse-alternative mappings for this location
  737: 			while (my ($left, $right) = each %alternatives) {
  738: 				if ($location =~ $right) {
  739: 					(my $l = $location) =~ s/$right/$left/;
  740: 					push @rc, $l;
  741: 				}
  742: 			}
  743: 		} else {
  744: 			return $location;
  745: 		}
  746: 	}
  747: 	return @rc;
  748: }

  806: # Purpose:  Resolve a full state/province name to its two-letter code using

  828: # Purpose:  Match parsed address components against all data rows.
  829: #           Each named column must match (case-insensitively) the corresponding
  830: #           data field; a minimum of 3 columns must match for a result.
  831: # Entry:    $self, $data — hashref of address components,
  832: #           @columns — ordered list of keys to compare.
  833: # Exit:     Geo::Location::Point on match; undef otherwise.
  834: # Side Effects: May delete undef entries from $data (for undefined optional fields).
  835: sub _search {
  836: 	my ($self, $data, @columns) = @_;
  837: 
  838: 	$self->{'reads'}++;
  839: 
  840: 	# M5 (boolean reduction): the $failed flag was set-and-break in the inner loop
  841: 	# then tested in the outer loop — eliminated by using a labeled 'next ROW'
  842: 	# that short-circuits directly, removing one boolean gate per iteration.
  843: 	ROW: for my $row (@{$self->{'data'}}) {
  844: 		my $matched = 0;
  845: 		my @matched_columns;
  846: 
  847: 		for my $column (@columns) {
  848: 			if (defined($data->{$column})) {
  849: 				next ROW unless defined($row->{$column});
  850: 				next ROW if uc($row->{$column}) ne uc($data->{$column});
  851: 				$matched++;
  852: 				# The accuracy is reported from the columns that matched, not
  853: 				# every column this call site offered: a caller trying
  854: 				# 'number road city...' for a town that has no road must not have
  855: 				# its match described as an address.
  856: 				push @matched_columns, $column;
  857: 			} elsif (exists $data->{$column}) {
  858: 				delete $data->{$column};
  859: 			}
  860: 		}
  861: 
  862: 		next if $matched < 3;
  863: 
  864: 		# Assign confidence based on how many columns contributed to the match
  865: 		my $confidence = $matched == scalar(@columns) ? $CONF_EXACT
  866: 		               : $matched >= 4                ? $CONF_HIGH
  867: 		               :                               $CONF_MEDIUM;

  869: 		# Provenance (Phase 6 dependency specs section 2.1).  Set after %{$row}
  870: 		# so that a dataset column that happens to be called accuracy cannot stand
  871: 		# in for the match this code actually made.

  874: 			confidence => $confidence,
  875: 			database   => __PACKAGE__,
  876: 			%{$row},
  877: 			geocoder   => GEOCODER,
  878: 			accuracy   => Geo::Location::Point::accuracy_for(@matched_columns),
  879: 		);
  880: 	}
  881: 	return;

  884: =head2 _result_for
  885: 
  886: Build the result for a dataset row that the index matched.  The index holds the
  887: rows themselves - the same hashrefs C<_search> scans - so the object is built
  888: as a copy rather than blessed in place, and the provenance keys every result
  889: carries (section 2.1 of F<BI-Phase6-Dependency-Specs.md>) are added where the
  890: row does not already record them.

  892: An index hit is an exact match on the row's string, so the confidence is
  893: C<$CONF_EXACT>, and the accuracy is what the row's own fields justify, through
  894: C<Geo::Location::Point::accuracy_for()>.

  910: # Purpose:  Calculate the geographic centres (centroids) of all city/state/country
  911: #           clusters in the parsed dataset that contain three or more data points.
  912: #           Clusters with fewer than 3 entries are too sparse to be meaningful.
  913: # Entry:    $rows — arrayref of hashrefs with latitude/longitude/city/state/country.
  914: # Exit:     Arrayref of hashrefs (one per qualifying cluster), or undef if none.
  915: # Side Effects: None.
  916: sub _find_geographic_centres {
  917: 	my $rows = $_[0];
  918: 
  919: 	# Group rows by normalised city|state|country key
  920: 	my %groups;
  921: 	for my $row (@{$rows}) {
  922: 		next unless defined($row->{'latitude'})  && defined($row->{'longitude'});
  923: 		next unless $row->{'latitude'}  =~ /^-?\d+\.?\d*$/;
  924: 		next unless $row->{'longitude'} =~ /^-?\d+\.?\d*$/;
  925: 		my $key = join '|',
  926: 			$row->{'city'}    // '',
  927: 			$row->{'state'}   // '',
  928: 			$row->{'country'} // '';
  929: 		push @{$groups{$key}}, $row;
  930: 	}
  931: 
  932: 	my @centres;
  933: 	for my $key (keys %groups) {
  934: 		my $locs = $groups{$key};
  935: 		next if @{$locs} < 3;
  936: 
  937: 		my ($city, $state, $country) = split /\|/, $key;
  938: 		my ($lat, $lon) = _calculate_centre($locs);
  939: 
  940: 		push @centres, {
  941: 			city      => $city,
  942: 			state     => $state,
  943: 			country   => $country,
  944: 			lat       => $lat,
  945: 			latitude  => $lat,
  946: 			longitude => $lon,
  947: 			long      => $lon,
  948: 			lng       => $lon,
  949: 		};
  950: 	}
  951: 
  952: 	return @centres ? \@centres : undef;
  953: }

  955: # Purpose:  Compute the arithmetic mean of latitude and longitude for a group
  956: #           of locations.  Accurate for small geographic areas; for large areas
  957: #           or locations crossing the antimeridian, a vector-mean is needed.
  958: # Entry:    $locs — arrayref of hashrefs each with 'latitude' and 'longitude'.
  959: # Exit:     ($centre_lat, $centre_lon) as decimal degree strings to 6dp.
  960: # Side Effects: None.
  961: sub _calculate_centre {
  962: 	my $locs  = $_[0];
  963: 	my ($sum_lat, $sum_lon) = (0, 0);
  964: 	$sum_lat += $_->{'latitude'}, $sum_lon += $_->{'longitude'} for @{$locs};
  965: 	my $n = scalar @{$locs};
  966: 	return (sprintf('%.6f', $sum_lat / $n), sprintf('%.6f', $sum_lon / $n));
  967: }

  971: sub _equal {
  972: 	my ($A, $B, $dp) = @_;
  973: 	return sprintf("%.${dp}g", $A) eq sprintf("%.${dp}g", $B);
